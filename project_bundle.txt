import 'dart:math' as math;
import 'onnx_service.dart';

enum Verdict { realMetal, possibleMetal, realVoid, possibleVoid, looseSoil, reliable, possible, uncertain }

/// One detected target (from the YOLO model and/or the rule-based detector).
class Finding {
  final String kind; // soil | void | moisture | organic | rock | metal
  final String source; // 'AI' or 'Rule'
  final double confidence; // 0..1
  final double r0, c0, r1, c1; // cell bounds in ORIGINAL grid units (r1/c1 exclusive)
  final double peakDelta; // signed extreme deviation from the normal level (counts)
  final double widthCm, lengthCm, depthM;
  final Verdict verdict;
  const Finding(this.kind, this.source, this.confidence, this.r0, this.c0, this.r1, this.c1,
      this.peakDelta, this.widthCm, this.lengthCm, this.depthM, this.verdict);
}

double _median(List<double> v) {
  final s = [...v]..sort();
  final n = s.length;
  return n.isOdd ? s[n ~/ 2] : (s[n ~/ 2 - 1] + s[n ~/ 2]) / 2;
}

/// Combines the YOLO detections (if the model ran) with a rule-based detector.
/// The rule-based part always works, so the analysis never returns empty-handed
/// just because the neural network failed to load.
List<Finding> buildFindings({
  required List<List<double>> m, // raw values, no NaN
  required int spacingCm,
  required double thrCounts, // user threshold converted to sensor counts
  required double soilFactor,
  AiResult? yolo,
}) {
  final rows = m.length, cols = m.first.length;
  final flat = m.expand((r) => r).toList();
  final normal = _median(flat); // "normal soil value"
  final sigma = 1.4826 * _median(flat.map((v) => (v - normal).abs()).toList());
  final thr = math.max(thrCounts, 3 * sigma).clamp(1.0, 1e9).toDouble();

  double depthAt(int pr, int pc) {
    final peak = m[pr][pc] - normal;
    final sign = peak >= 0 ? 1.0 : -1.0;
    final half = peak.abs() / 2;
    int run(int dr, int dc) {
      var n = 0, r = pr + dr, c = pc + dc;
      while (r >= 0 && r < rows && c >= 0 && c < cols && sign * (m[r][c] - normal) >= half) {
        n++;
        r += dr;
        c += dc;
      }
      return n;
    }

    final w = ((run(0, -1) + run(0, 1) + 1) + (run(-1, 0) + run(1, 0) + 1)) / 2;
    final fwhm = w * spacingCm / 100; // metres
    return (fwhm / 1.5 * soilFactor).clamp(0.1, 5.0).toDouble(); // rough dipole rule of thumb
  }

  Verdict verdictFor(String kind, double conf, double peak) {
    final okMetal = peak > 0.8 * thr, okVoid = peak < -0.8 * thr;
    if (kind == 'metal') {
      if (okMetal) return conf >= 0.6 ? Verdict.realMetal : Verdict.possibleMetal;
      return conf >= 0.35 ? Verdict.looseSoil : Verdict.uncertain;
    }
    if (kind == 'void') {
      if (okVoid) return conf >= 0.6 ? Verdict.realVoid : Verdict.possibleVoid;
      return conf >= 0.35 ? Verdict.looseSoil : Verdict.uncertain;
    }
    return conf >= 0.6 ? Verdict.reliable : (conf >= 0.4 ? Verdict.possible : Verdict.uncertain);
  }

  final out = <Finding>[];

  // ---- 1) YOLO detections -> cell coordinates --------------------------
  if (yolo != null) {
    final scale = math.min(640 / cols, 640 / rows);
    final x0 = (640 - cols * scale) / 2, y0 = (640 - rows * scale) / 2;
    for (final d in yolo.detections.take(12)) {
      final c0 = ((d.x1 - x0) / scale).clamp(0.0, cols - 1.0).toDouble();
      final r0 = ((d.y1 - y0) / scale).clamp(0.0, rows - 1.0).toDouble();
      final c1 = ((d.x2 - x0) / scale).clamp(c0 + 1, cols.toDouble()).toDouble();
      final r1 = ((d.y2 - y0) / scale).clamp(r0 + 1, rows.toDouble()).toDouble();
      var peak = 0.0, pr = r0.floor(), pc = c0.floor();
      for (var r = r0.floor(); r < r1.ceil() && r < rows; r++) {
        for (var c = c0.floor(); c < c1.ceil() && c < cols; c++) {
          final dv = m[r][c] - normal;
          if (dv.abs() > peak.abs()) {
            peak = dv;
            pr = r;
            pc = c;
          }
        }
      }
      out.add(Finding(d.label, 'AI', d.confidence, r0, c0, r1, c1, peak,
          (c1 - c0) * spacingCm, (r1 - r0) * spacingCm, depthAt(pr, pc),
          verdictFor(d.label, d.confidence, peak)));
    }
  }

  // ---- 2) rule-based detector: connected components over/under normal ----
  final seen = List.generate(rows, (_) => List.filled(cols, false));
  for (var r = 0; r < rows; r++) {
    for (var c = 0; c < cols; c++) {
      final dv = m[r][c] - normal;
      if (seen[r][c] || dv.abs() <= thr) continue;
      final sign = dv > 0 ? 1.0 : -1.0;
      final stack = [(r, c)];
      seen[r][c] = true;
      var rMin = r, rMax = r, cMin = c, cMax = c, area = 0, pr = r, pc = c;
      var peak = dv;
      while (stack.isNotEmpty) {
        final (cr, cc) = stack.removeLast();
        area++;
        rMin = math.min(rMin, cr);
        rMax = math.max(rMax, cr);
        cMin = math.min(cMin, cc);
        cMax = math.max(cMax, cc);
        final v = m[cr][cc] - normal;
        if (v.abs() > peak.abs()) {
          peak = v;
          pr = cr;
          pc = cc;
        }
        for (final (dr, dc) in [(1, 0), (-1, 0), (0, 1), (0, -1)]) {
          final nr = cr + dr, nc = cc + dc;
          if (nr < 0 || nr >= rows || nc < 0 || nc >= cols || seen[nr][nc]) continue;
          final nv = m[nr][nc] - normal;
          if (nv * sign > thr) {
            seen[nr][nc] = true;
            stack.add((nr, nc));
          }
        }
      }
      final ratio = peak.abs() / thr;
      if (area == 1 && ratio < 2) continue; // lone weak cell = noise
      final compact = ratio >= 2 || area <= 4;
      final kind = sign > 0 ? (compact ? 'metal' : 'rock') : (compact ? 'void' : 'moisture');
      var conf = math.min(0.9, ratio / (ratio + 1.5));
      if (!compact) conf *= 0.7;
      final f = Finding(kind, 'Rule', conf, rMin.toDouble(), cMin.toDouble(), rMax + 1.0, cMax + 1.0,
          peak, (cMax - cMin + 1) * spacingCm.toDouble(), (rMax - rMin + 1) * spacingCm.toDouble(),
          depthAt(pr, pc), verdictFor(kind, conf, peak));
      // skip if an AI box already covers this spot
      final dup = out.any((o) => o.source == 'AI' && _iou(o, f) > 0.25);
      if (!dup) out.add(f);
    }
  }

  out.sort((a, b) => b.confidence.compareTo(a.confidence));
  return out;
}

double _iou(Finding a, Finding b) {
  final ix = math.max(0.0, math.min(a.c1, b.c1) - math.max(a.c0, b.c0));
  final iy = math.max(0.0, math.min(a.r1, b.r1) - math.max(a.r0, b.r0));
  final inter = ix * iy;
  final u = (a.c1 - a.c0) * (a.r1 - a.r0) + (b.c1 - b.c0) * (b.r1 - b.r0) - inter;
  return u <= 0 ? 0 : inter / u;
}

import 'dart:typed_data';
import '../features/analysis_3d/grid_processor.dart' show ScanStats, jetRelative;

/// Matrix (RAW sensor values, no NaN) -> YOLO tensor float32 NCHW [1,3,640,640], RGB 0..1.
/// Pipeline: optional plane detrend -> bicubic upsampling -> data-relative Jet colours
/// -> letterbox (grey 114/255), matching Ultralytics preprocessing.
class HeatmapRenderer {
  static const int size = 640;

  static Float32List toTensor(List<List<double>> raw, {bool detrend = true, bool imagenet = false}) {
    var m = detrend ? removePlane(raw) : raw;
    final rows = m.length, cols = m.first.length;

    // Statistics AFTER detrending, because colours are relative to them.
    final flat = m.expand((r) => r).toList();
    final lo = flat.reduce((a, b) => a < b ? a : b);
    final hi = flat.reduce((a, b) => a > b ? a : b);
    final avg = flat.reduce((a, b) => a + b) / flat.length;
    final stats = ScanStats(lo, avg, hi);

    final scale = (size / cols) < (size / rows) ? size / cols : size / rows;
    final w = (cols * scale).round(), h = (rows * scale).round();
    final x0 = (size - w) ~/ 2, y0 = (size - h) ~/ 2;

    const pad = 114 / 255.0;
    const mean = [0.485, 0.456, 0.406], std = [0.229, 0.224, 0.225];
    double nrm(double v, int ch) => imagenet ? (v - mean[ch]) / std[ch] : v;
    final plane = size * size;
    final out = Float32List(3 * plane);
    for (var ch = 0; ch < 3; ch++) {
      out.fillRange(ch * plane, (ch + 1) * plane, nrm(pad, ch));
    }

    for (var y = 0; y < h; y++) {
      final fy = ((y + 0.5) / scale - 0.5).clamp(0.0, rows - 1.0);
      for (var x = 0; x < w; x++) {
        final fx = ((x + 0.5) / scale - 0.5).clamp(0.0, cols - 1.0);
        final v = bicubic(m, fy, fx).clamp(lo, hi); // clamp kills overshoot
        final (r, g, b) = jetRelative(v, stats);
        final p = (y0 + y) * size + (x0 + x);
        out[p] = nrm(r, 0);
        out[plane + p] = nrm(g, 1);
        out[2 * plane + p] = nrm(b, 2);
      }
    }
    return out;
  }

  /// Least-squares plane z = a*x + b*y + c is subtracted (the mean is kept),
  /// removing a uniform tilt/gradient so local anomalies stand out.
  static List<List<double>> removePlane(List<List<double>> m) {
    final rows = m.length, cols = m.first.length;
    final xm = (cols - 1) / 2, ym = (rows - 1) / 2;
    double mean = 0, sxz = 0, syz = 0, sxx = 0, syy = 0;
    for (var r = 0; r < rows; r++) {
      for (var c = 0; c < cols; c++) {
        mean += m[r][c];
      }
    }
    mean /= rows * cols;
    for (var r = 0; r < rows; r++) {
      for (var c = 0; c < cols; c++) {
        final dx = c - xm, dy = r - ym, dz = m[r][c] - mean;
        sxz += dx * dz; syz += dy * dz; sxx += dx * dx; syy += dy * dy;
      }
    }
    final a = sxx == 0 ? 0.0 : sxz / sxx, b = syy == 0 ? 0.0 : syz / syy;
    return [
      for (var r = 0; r < rows; r++)
        [for (var c = 0; c < cols; c++) m[r][c] - a * (c - xm) - b * (r - ym)]
    ];
  }

  // Catmull-Rom cubic interpolation on a 4x4 neighbourhood.
  static double _cr(double t, double p0, double p1, double p2, double p3) =>
      0.5 * ((2 * p1) + (-p0 + p2) * t + (2 * p0 - 5 * p1 + 4 * p2 - p3) * t * t +
          (-p0 + 3 * p1 - 3 * p2 + p3) * t * t * t);

  static double bicubic(List<List<double>> m, double fy, double fx) {
    final rows = m.length, cols = m.first.length;
    final iy = fy.floor(), ix = fx.floor();
    final ty = fy - iy, tx = fx - ix;
    final col = List<double>.filled(4, 0);
    for (var j = -1; j <= 2; j++) {
      final r = (iy + j).clamp(0, rows - 1);
      double g(int dx) => m[r][(ix + dx).clamp(0, cols - 1)];
      col[j + 1] = _cr(tx, g(-1), g(0), g(1), g(2));
    }
    return _cr(ty, col[0], col[1], col[2], col[3]);
  }
}

/// One detected anomaly (box in 640x640 letterboxed pixel space).
class Detection {
  final int classId;
  final String label;
  final double confidence; // 0..1
  final double x1, y1, x2, y2;
  const Detection(this.classId, this.label, this.confidence, this.x1, this.y1, this.x2, this.y2);
}

class AiResult {
  final List<Detection> detections; // after NMS, sorted by confidence
  final Map<String, double> bestPerClass; // best score per class, over ALL anchors (before threshold)
  final double maxScore; // highest class score anywhere: tells "ran, nothing found" from "broken"
  final int millis; // inference time
  const AiResult(this.detections, this.bestPerClass, {this.maxScore = 0, this.millis = 0});
  Detection? get top => detections.isEmpty ? null : detections.first;
}

class OnnxServiceException implements Exception {
  final String message;
  OnnxServiceException(this.message);
  @override
  String toString() => 'OnnxServiceException: $message';
}

/// Local inference for best2.onnx (Ultralytics YOLOv8 detect, 6 classes).
///   input  'images'  float32 [1,3,640,640]  RGB 0..1, NCHW
///   output 'output0' float32 [1,10,8400]    channel-major: value(c, i) = flat[c * 8400 + i]
///                    c 0-3 = cx,cy,w,h (640-px letterbox space), c 4-9 = class scores (already sigmoid)
///   NMS is NOT inside the graph, so it is done here.
class OnnxService {
  static const modelAsset = 'assets/models/best2.onnx';
  static const inputName = 'images';
  static const outputName = 'output0';
  static const labels = ['soil', 'void', 'moisture', 'organic', 'rock', 'metal'];
  static const numAnchors = 8400;
  static const size = HeatmapRenderer.size; // 640
  static const _maxCandidates = 300; // cap before NMS (keeps NMS fast)

  OrtSession? _session;
  bool _busy = false;
  bool get isReady => _session != null;

  Future<void> init() async {
    if (_session != null) return;
    try {
      await CrashGuard.mark('init: createSession');
      _session = await OnnxRuntime().createSessionFromAsset(modelAsset /*ORT_OPTS*/);
      await CrashGuard.mark('done');
    } catch (e) {
      await CrashGuard.mark('done');
      throw OnnxServiceException('Failed to load model: $e');
    }
  }

  /// [raw] = 2-D matrix of RAW sensor values without NaN (GridProcessor.filled()).
  Future<AiResult> analyze(
    List<List<double>> raw, {
    double confThreshold = 0.25,
    double iouThreshold = 0.45,
    bool detrend = true,
    bool imagenet = false,
  }) async {
    if (_session == null) throw OnnxServiceException('Call init() first.');
    if (_busy) throw OnnxServiceException('Inference already running.');
    if (raw.isEmpty || raw.first.isEmpty) throw OnnxServiceException('Empty matrix.');
    _busy = true;

    final sw = Stopwatch()..start();
    OrtValue? input;
    Map<String, OrtValue>? outputs;
    try {
      // 1) matrix -> heat-map -> flat Float32List, NCHW (heavy loop: off the UI isolate)
      await CrashGuard.mark('infer: buildTensor');
      final Float32List tensor = await compute(_buildTensor, (raw, detrend, imagenet));
      const expected = 1 * 3 * size * size;
      if (tensor.length != expected) {
        throw OnnxServiceException('Tensor length ${tensor.length}, expected $expected.');
      }
      for (var i = 0; i < tensor.length; i += 997) {
        if (!tensor[i].isFinite) throw OnnxServiceException('Tensor contains NaN/Inf.');
      }

      // 2) shape [1,3,640,640]
      await CrashGuard.mark('infer: createInputValue');
      input = await OrtValue.fromList(tensor, [1, 3, size, size]);

      // 3) run
      await CrashGuard.mark('infer: session.run');
      outputs = await _session!.run({inputName: input});
      final out = outputs[outputName] ?? (outputs.isEmpty ? null : outputs.values.first);
      if (out == null) throw OnnxServiceException('Model returned no outputs.');

      // 4) read: flatten whatever nesting the plugin returns, convert safely to double
      await CrashGuard.mark('infer: readOutput');
      final rawOut = await out.asFlattenedList();
      const want = (4 + 6) * numAnchors;
      if (rawOut.length != want) {
        throw OnnxServiceException('Unexpected output size ${rawOut.length} (expected $want).');
      }
      final flat = Float32List(want);
      for (var i = 0; i < want; i++) {
        final e = rawOut[i];
        flat[i] = e is num ? e.toDouble() : double.tryParse('$e') ?? double.nan;
      }

      // 5) decode + NMS
      await CrashGuard.mark('infer: decode');
      final res = _decode(flat, confThreshold, iouThreshold, sw.elapsedMilliseconds);
      await CrashGuard.mark('done');
      return res;
    } on OnnxServiceException {
      rethrow;
    } catch (e) {
      throw OnnxServiceException('Inference failed: $e');
    } finally {
      // Always free native memory, even on error.
      await CrashGuard.mark('infer: dispose');
      try {
        await input?.dispose();
        if (outputs != null) {
          for (final v in outputs.values) {
            await v.dispose();
          }
        }
      } catch (_) {}
      await CrashGuard.mark('done');
      _busy = false;
    }
  }

  Future<void> dispose() async {
    await _session?.close();
    _session = null;
  }

  // ---- helpers ---------------------------------------------------------
  static Float32List _buildTensor((List<List<double>>, bool, bool) a) =>
      HeatmapRenderer.toTensor(a.$1, detrend: a.$2, imagenet: a.$3);

  static AiResult _decode(Float32List f, double conf, double iou, int ms) {
    final cand = <Detection>[];
    final best = {for (final l in labels) l: 0.0};
    var maxScore = 0.0;
    for (var i = 0; i < numAnchors; i++) {
      var cls = 0;
      var score = f[4 * numAnchors + i];
      for (var c = 1; c < labels.length; c++) {
        final s = f[(4 + c) * numAnchors + i];
        if (s > score) {
          score = s;
          cls = c;
        }
      }
      if (score.isNaN) continue;
      if (score > maxScore) maxScore = score;
      if (score > best[labels[cls]]!) best[labels[cls]] = score;
      if (score < conf) continue;
      final cx = f[i], cy = f[numAnchors + i];
      final w = f[2 * numAnchors + i], h = f[3 * numAnchors + i];
      cand.add(Detection(cls, labels[cls], score, cx - w / 2, cy - h / 2, cx + w / 2, cy + h / 2));
    }
    cand.sort((a, b) => b.confidence.compareTo(a.confidence));
    final top = cand.length > _maxCandidates ? cand.sublist(0, _maxCandidates) : cand;
    return AiResult(_nms(top, iou), best, maxScore: maxScore, millis: ms);
  }

  /// Per-class greedy NMS (input already sorted by confidence).
  static List<Detection> _nms(List<Detection> d, double iouThr) {
    final keep = <Detection>[];
    for (final c in d) {
      if (keep.every((k) => k.classId != c.classId || _iou(k, c) < iouThr)) keep.add(c);
      if (keep.length >= 50) break;
    }
    return keep;
  }

  static double _iou(Detection a, Detection b) {
    final ix = math.max(0.0, math.min(a.x2, b.x2) - math.max(a.x1, b.x1));
    final iy = math.max(0.0, math.min(a.y2, b.y2) - math.max(a.y1, b.y1));
    final inter = ix * iy;
    final u = (a.x2 - a.x1) * (a.y2 - a.y1) + (b.x2 - b.x1) * (b.y2 - b.y1) - inter;
    return u <= 0 ? 0 : inter / u;
  }
}

import 'dart:math' as math;
import '../../ai/heatmap_renderer.dart';

/// The five filter chips of the 3D/4D screen.
class ScanFilters {
  bool errorCorrection = false; // clip outliers (mean +- 2.5 sigma)
  bool signalCorrection = false; // remove tilt (plane detrend)
  bool interpolation = false; // bicubic x2
  bool subdivision = false; // bilinear x2 (finer cells)
  int rotation = 0; // quarter turns
}

/// Returns (processed matrix, upsampling factor).
(List<List<double>>, int) applyFilters(List<List<double>> input, ScanFilters f) {
  var m = input;
  var factor = 1;
  if (f.errorCorrection) m = _clip(m);
  if (f.signalCorrection) m = HeatmapRenderer.removePlane(m);
  if (f.interpolation) {
    m = _upsample(m, 2, true);
    factor = 2;
  } else if (f.subdivision) {
    m = _upsample(m, 2, false);
    factor = 2;
  }
  for (var i = 0; i < f.rotation % 4; i++) {
    m = _rot90(m);
  }
  return (m, factor);
}

List<List<double>> _clip(List<List<double>> m) {
  final flat = m.expand((r) => r).toList();
  final mean = flat.reduce((a, b) => a + b) / flat.length;
  final sd = math.sqrt(flat.map((v) => (v - mean) * (v - mean)).reduce((a, b) => a + b) / flat.length);
  final lo = mean - 2.5 * sd, hi = mean + 2.5 * sd;
  return [for (final r in m) [for (final v in r) v.clamp(lo, hi).toDouble()]];
}

List<List<double>> _upsample(List<List<double>> m, int k, bool cubic) {
  final rows = m.length, cols = m.first.length;
  return List.generate(rows * k, (y) {
    final fy = ((y + 0.5) / k - 0.5).clamp(0.0, rows - 1.0).toDouble();
    return List.generate(cols * k, (x) {
      final fx = ((x + 0.5) / k - 0.5).clamp(0.0, cols - 1.0).toDouble();
      if (cubic) return HeatmapRenderer.bicubic(m, fy, fx);
      final iy = fy.floor(), ix = fx.floor();
      final iy1 = math.min(iy + 1, rows - 1), ix1 = math.min(ix + 1, cols - 1);
      final ty = fy - iy, tx = fx - ix;
      return (m[iy][ix] * (1 - tx) + m[iy][ix1] * tx) * (1 - ty) +
          (m[iy1][ix] * (1 - tx) + m[iy1][ix1] * tx) * ty;
    });
  });
}

List<List<double>> _rot90(List<List<double>> m) {
  final rows = m.length, cols = m.first.length;
  return List.generate(cols, (c) => List.generate(rows, (r) => m[rows - 1 - r][c]));
}

/// Data-relative 5-anchor Jet palette. Returns r,g,b in 0..1.
(double, double, double) jetRelative(double v, ScanStats s) {
  final a = [s.min, (s.min + s.avg) / 2, s.avg, (s.avg + s.max) / 2, s.max];
  const cols = [
    (0.0, 0.0, 1.0),
    (0.0, 1.0, 1.0),
    (0.0, 1.0, 0.0),
    (1.0, 1.0, 0.0),
    (1.0, 0.0, 0.0),
  ];
  if (v <= a.first) return cols.first;
  if (v >= a.last) return cols.last;
  for (var i = 1; i < a.length; i++) {
    if (v < a[i]) {
      final span = a[i] - a[i - 1];
      final t = span <= 1e-9 ? 0.0 : (v - a[i - 1]) / span;
      final (r0, g0, b0) = cols[i - 1];
      final (r1, g1, b1) = cols[i];
      return (r0 + (r1 - r0) * t, g0 + (g1 - g0) * t, b0 + (b1 - b0) * t);
    }
  }
  return cols.last;
}

Color jetRelativeColor(double v, ScanStats s) {
  final (r, g, b) = jetRelative(v, s);
  return Color.fromRGBO((r * 255).round(), (g * 255).round(), (b * 255).round(), 1);
}

/// Which display layer a value belongs to (nearest Jet anchor).
Layer layerOf(double v, ScanStats s) {
  final a = [s.min, (s.min + s.avg) / 2, s.avg, (s.avg + s.max) / 2, s.max];
  if (s.max - s.min < 1e-9) return Layer.soil;
  if (v < (a[0] + a[1]) / 2) return Layer.voids;
  if (v < (a[1] + a[2]) / 2) return Layer.moisture;
  if (v < (a[2] + a[3]) / 2) return Layer.soil;
  if (v < (a[3] + a[4]) / 2) return Layer.minerals;
  return Layer.metals;
}

/// App-wide singleton. Only the locale uses ChangeNotifier (root rebuild);
/// everything that changes often uses ValueNotifier to keep rebuilds local.
class AppState extends ChangeNotifier {
  AppState._() {
    ble.samples.listen(_onSample);
    _checkCrash();
    grid.addListener(_persist);
    _restore();
  }

  // ---- the current scan survives app restarts / crashes ----------------
  Timer? _saveT;
  Future<File> _curFile() async =>
      File('${(await getApplicationSupportDirectory()).path}/current_scan.json');

  void _persist() {
    _saveT?.cancel();
    _saveT = Timer(const Duration(milliseconds: 500), () async {
      try {
        await (await _curFile()).writeAsString(jsonEncode(grid.toJson('current')));
      } catch (_) {}
    });
  }

  Future<void> _restore() async {
    try {
      final f = await _curFile();
      if (!await f.exists()) return;
      final j = jsonDecode(await f.readAsString()) as Map<String, dynamic>;
      final m = [
        for (final r in j['matrix'] as List)
          [for (final v in r as List) v == null ? double.nan : (v as num).toDouble()]
      ];
      grid.snake = j['snake'] as bool? ?? grid.snake;
      grid.startRight = j['start_right'] as bool? ?? grid.startRight;
      grid.loadMatrix(m, spacing: j['spacing_cm'] as int?, cursor: j['cursor'] as int?);
    } catch (_) {}
  }

  /// If the last AI run died natively, switch the neural model off and remember where.
  Future<void> _checkCrash() async {
    final st = await CrashGuard.previousCrashStage();
    if (st != null) {
      aiEnabled.value = false;
      crashStage.value = st;
      await CrashGuard.mark('done');
    }
  }
  static final I = AppState._();

  final ble = BleService();
  final grid = GridProcessor(rows: 16, cols: 16, spacingCm: 30);
  final onnx = OnnxService();

  Locale locale = const Locale('en');
  void toggleLocale() {
    locale = locale.languageCode == 'en' ? const Locale('ar') : const Locale('en');

/// BLE link to the ESP32 gradiometer.
/// Protocol: ASCII lines "0..1023\r\n". Optional "BAT:87\r\n" for battery.
class BleService {
  // Nordic UART defaults. If your firmware uses other UUIDs, the service falls back
  // to the first characteristic that supports notifications.
  static final Guid txCharUuid = Guid('6E400003-B5A3-F393-E0A9-E50E24DCCA9E');

  final _samples = StreamController<SensorSample>.broadcast();
  final _status = StreamController<BleStatus>.broadcast();
  Stream<SensorSample> get samples => _samples.stream;
  Stream<BleStatus> get status => _status.stream;

  BleStatus current = BleStatus.disconnected;
  final ValueNotifier<int> battery = ValueNotifier<int>(-1);
  String? connectedName;
  String? lastName;

  BluetoothDevice? _device;
  StreamSubscription? _connSub, _dataSub;
  String _buffer = '';
  bool _wantConnected = false;
  int _retry = 0;

  void _emit(BleStatus s) {
    current = s;
    _status.add(s);
  }

  /// Used by the built-in sensor simulator.
  void injectSample(int v) => _samples.add(SensorSample(v, DateTime.now()));

  // ---- scanning --------------------------------------------------------
  Stream<List<ScanResult>> get results => FlutterBluePlus.scanResults;
  Stream<bool> get scanning => FlutterBluePlus.isScanning;
  Future<void> startScan() =>
      FlutterBluePlus.startScan(timeout: const Duration(seconds: 8));

  // ---- connect / auto-reconnect ---------------------------------------
  Future<void> connect(BluetoothDevice device) async {
    _wantConnected = true;
    _device = device;
    connectedName = device.platformName.isEmpty ? device.remoteId.str : device.platformName;
    lastName = connectedName;
    await _saveLastDevice(device.remoteId.str, connectedName!);
    await _openLink();
  }

  Future<void> autoConnectLast() async {
    final info = await lastDevice();
    if (info == null) return;
    lastName = info['name'];
    await connect(BluetoothDevice.fromId(info['id']!));
  }

  Future<void> _openLink() async {
    final d = _device;
    if (d == null) return;
    try {
      _emit(BleStatus.connecting);
      await FlutterBluePlus.stopScan();
      await d.connect(timeout: const Duration(seconds: 10), autoConnect: false);

      await _connSub?.cancel();
      _connSub = d.connectionState.listen((s) {
        if (s == BluetoothConnectionState.disconnected && _wantConnected) {
          _emit(BleStatus.reconnecting);
          _scheduleReconnect();
        }
      });

      final services = await d.discoverServices();
      BluetoothCharacteristic? ch;
      for (final s in services) {
        for (final c in s.characteristics) {
          if (c.uuid == txCharUuid) ch = c;
        }
      }
      ch ??= services
          .expand((s) => s.characteristics)
          .where((c) => c.properties.notify)
          .firstOrNull;
      if (ch == null) throw Exception('No notify characteristic');
      await ch.setNotifyValue(true);

      await _dataSub?.cancel();
      _dataSub = ch.onValueReceived.listen(_onBytes);
      d.cancelWhenDisconnected(_dataSub!);

      _retry = 0;
      _buffer = '';
      _emit(BleStatus.connected);
    } catch (_) {
      _emit(BleStatus.reconnecting);
      _scheduleReconnect();
    }
  }

  void _scheduleReconnect() {
    if (!_wantConnected) return;
    final delay = Duration(seconds: (1 << _retry.clamp(0, 5)));
    _retry++;
    Timer(delay, () {
      if (_wantConnected) _openLink();
    });
  }

  Future<void> disconnect() async {
    _wantConnected = false;
    await _dataSub?.cancel();
    await _connSub?.cancel();
    await _device?.disconnect();
    _emit(BleStatus.disconnected);
  }

  // ---- parsing ---------------------------------------------------------
  void _onBytes(List<int> bytes) {
    _buffer += utf8.decode(bytes, allowMalformed: true);
    int nl;
    while ((nl = _buffer.indexOf('\n')) >= 0) {
      final line = _buffer.substring(0, nl).trim();
      _buffer = _buffer.substring(nl + 1);
      if (line.isEmpty) continue;
      if (line.startsWith('BAT:')) {
        battery.value = int.tryParse(line.substring(4)) ?? battery.value;
        continue;
      }
      final v = int.tryParse(line);
      if (v != null && v >= 0 && v <= 1023) injectSample(v);
    }
  }

  // ---- last device (JSON file) ----------------------------------------
  Future<File> _cfg() async =>
      File('${(await getApplicationSupportDirectory()).path}/last_device.json');
  Future<void> _saveLastDevice(String id, String name) async =>
      (await _cfg()).writeAsString(jsonEncode({'id': id, 'name': name}));
  Future<Map<String, String>?> lastDevice() async {
    try {

      final m = jsonDecode(await f.readAsString()) as Map;
      if (m['id'] == null) return null;
      return {'id': m['id'] as String, 'name': (m['name'] ?? m['id']) as String};
    } catch (_) {
      return null;
    }
  }

class TonePlayer {
  final AudioPlayer _p = AudioPlayer();
  late final Uint8List _metal = _wav([[1500, 0.09], [0, 0.05], [1500, 0.09]]);
  late final Uint8List _void = _wav([[320, 0.30]]);

  Future<void> metal() => _play(_metal);
  Future<void> voidTone() => _play(_void);

  Future<void> _play(Uint8List b) async {
    try {
      await _p.stop();
      await _p.play(BytesSource(b));
    } catch (_) {/* audio is best-effort */}
  }

  /// segments: [frequencyHz (0 = silence), seconds]. 16-bit mono PCM WAV, 22.05 kHz.
  static Uint8List _wav(List<List<double>> segs) {
    const rate = 22050;
    final samples = <int>[];
    for (final s in segs) {
      final n = (s[1] * rate).round();
      for (var i = 0; i < n; i++) {
        final fade = math.min(1.0, math.min(i, n - i) / (rate * 0.008)); // click-free edges
        final v = s[0] == 0 ? 0.0 : math.sin(2 * math.pi * s[0] * i / rate) * 0.8 * fade;
        samples.add((v * 32767).round());
      }
    }
    final data = ByteData(44 + samples.length * 2);
    void str(int o, String t) {
      for (var i = 0; i < t.length; i++) {
        data.setUint8(o + i, t.codeUnitAt(i));
      }
    }

    str(0, 'RIFF');
    data.setUint32(4, 36 + samples.length * 2, Endian.little);
    str(8, 'WAVE');
    str(12, 'fmt ');
    data.setUint32(16, 16, Endian.little);
    data.setUint16(20, 1, Endian.little);
    data.setUint16(22, 1, Endian.little);
    data.setUint32(24, rate, Endian.little);
    data.setUint32(28, rate * 2, Endian.little);
    data.setUint16(32, 2, Endian.little);
    data.setUint16(34, 16, Endian.little);
    str(36, 'data');
    data.setUint32(40, samples.length * 2, Endian.little);
    for (var i = 0; i < samples.length; i++) {
      data.setInt16(44 + i * 2, samples[i], Endian.little);
    }
    return data.buffer.asUint8List();

    final stats = ScanStats(lo, flat.reduce((a, b) => a + b) / flat.length, hi);
    final span = (hi - lo).abs() < 1e-9 ? 1.0 : hi - lo;

    final cr = math.cos(rotation), sr = math.sin(rotation);
    Offset pu(double x, double y, double z) {
      final rx = x * cr - y * sr, ry = x * sr + y * cr;
      return Offset((rx - ry) * 0.866, (rx + ry) * 0.5 - z);
    }

    final maxH = 5.0 * depth;
    var minX = 1e9, maxX = -1e9, minY = 1e9, maxY = -1e9;
    for (final x in [-cols / 2, cols / 2]) {
      for (final y in [-rows / 2, rows / 2]) {
        for (final z in [0.0, maxH]) {
          final p = pu(x, y, z);
          minX = math.min(minX, p.dx);
          maxX = math.max(maxX, p.dx);
          minY = math.min(minY, p.dy);
          maxY = math.max(maxY, p.dy);
        }
      }
    }
    final scale = math.min(size.width * 0.94 / (maxX - minX), size.height * 0.92 / (maxY - minY));
    final bc = Offset((minX + maxX) / 2, (minY + maxY) / 2);
    final center = Offset(size.width / 2, size.height / 2);
    Offset P(double x, double y, double z) => center + (pu(x, y, z) - bc) * scale;

    // Floor outline
    final floor = Path()
      ..moveTo(P(-cols / 2, -rows / 2, 0).dx, P(-cols / 2, -rows / 2, 0).dy)
      ..lineTo(P(cols / 2, -rows / 2, 0).dx, P(cols / 2, -rows / 2, 0).dy)
      ..lineTo(P(cols / 2, rows / 2, 0).dx, P(cols / 2, rows / 2, 0).dy)
      ..lineTo(P(-cols / 2, rows / 2, 0).dx, P(-cols / 2, rows / 2, 0).dy)
      ..close();
    canvas.drawPath(floor, Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1
      ..color = TM.cyan.withOpacity(0.4));

    // Painter's order: far -> near (larger rx+ry is nearer to the viewer).
    final order = <List<num>>[];
    for (var r = 0; r < rows; r++) {
      for (var c = 0; c < cols; c++) {
        final x = c - cols / 2 + 0.5, y = r - rows / 2 + 0.5;
        order.add([r, c, (x * cr - y * sr) + (x * sr + y * cr)]);
      }
    }
    order.sort((a, b) => a[2].compareTo(b[2]));

    void quad(List<Offset> pts, Color color, {double stroke = 0, Color? sc}) {
      final path = Path()..addPolygon(pts, true);
      canvas.drawPath(path, Paint()..color = color);
      canvas.drawPath(path, Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = stroke == 0 ? 0.4 : stroke
        ..color = sc ?? Colors.black26);
    }

    for (final o in order) {
      final r = o[0].toInt(), c = o[1].toInt();
      final v = m[r][c];
      final hid = hidden.contains(layerOf(v, stats));
      final n = ((v - lo) / span).clamp(0.0, 1.0);
      final h = hid ? 0.1 : math.max(0.15, n * maxH);
      final base = hid
          ? Colors.blueGrey.withOpacity(0.18)
          : jetRelativeColor(v, stats).withOpacity(volumetric ? 0.4 : 1.0);
      final x = c - cols / 2, y = r - rows / 2;

      // [x1,y1,x2,y2,nx,ny] for the 4 side faces
      final sides = [
        [x, y, x + 1, y, 0.0, -1.0],
        [x + 1, y, x + 1, y + 1, 1.0, 0.0],
        [x + 1, y + 1, x, y + 1, 0.0, 1.0],
        [x, y + 1, x, y, -1.0, 0.0],
      ];
      for (var i = 0; i < 4; i++) {
        final s = sides[i];
        final nx = s[4], ny = s[5];
        final vis = (nx * cr - ny * sr) + (nx * sr + ny * cr) > 0;
        if (!vis) continue;
        quad([P(s[0], s[1], 0), P(s[2], s[3], 0), P(s[2], s[3], h), P(s[0], s[1], h)],
            Color.lerp(base, Colors.black, i.isEven ? 0.4 : 0.25)!);
      }
      final marked = highlight != null && !hid && highlight!.contains(Offset(c + 0.5, r + 0.5));
      quad([P(x, y, h), P(x + 1, y, h), P(x + 1, y + 1, h), P(x, y + 1, h)], base,
          stroke: marked ? 1.8 : 0, sc: marked ? TM.gold : null);
    }
  }

  @override
  bool shouldRepaint(covariant IsoGridPainter old) => true;
}

import sys, re, glob, os
P = sys.argv[1]
if not P or not os.path.isdir(P):
    print("plugin dir not found - skipping"); sys.exit(0)

files = glob.glob(P + '/lib/**/*.dart', recursive=True)
src = {f: open(f, encoding='utf-8', errors='ignore').read() for f in files}
alltxt = "\n".join(src.values())

def body_of(kind, name):
    m = re.search(kind + r'\s+' + name + r'\b[^{]*\{(.*?)\n\}', alltxt, re.S)
    return m.group(1) if m else ''

opts = body_of('class', 'OrtSessionOptions')
prov = body_of('enum', 'OrtProvider')
sigs = re.findall(r'createSessionFromAsset\s*\(([^)]*)\)', alltxt, re.S)
print("OrtSessionOptions fields:", re.findall(r'this\.(\w+)', opts))
print("OrtProvider values:", re.findall(r'\b([A-Za-z_]+)\s*[,;(]', prov)[:12])
for s in sigs: print("createSessionFromAsset(", ' '.join(s.split()), ")")

can_options = any(re.search(r'OrtSessionOptions\??\s+options', s) for s in sigs)
args = []
if re.search(r'this\.providers', opts) and re.search(r'\bCPU\b', prov):
    args.append('providers: [OrtProvider.CPU]')
if re.search(r'this\.intraOpNumThreads', opts):
    args.append('intraOpNumThreads: 2')

# the class must be exported from the package's main library
main = os.path.join(P, 'lib', 'flutter_onnxruntime.dart')
defining = [f for f, t in src.items() if re.search(r'class\s+OrtSessionOptions\b', t)]
exported = bool(defining) and (defining[0] == main or
           os.path.basename(defining[0]) in open(main, encoding='utf-8', errors='ignore').read())

target = 'lib/ai/onnx_service.dart'
t = open(target, encoding='utf-8').read()

    t = t.replace('/*ORT_OPTS*/', ', options: OrtSessionOptions(' + ', '.join(args) + ')')
    open(target, 'w', encoding='utf-8').write(t)
    print("PATCHED session options:", args)
else:
    print("NOT patched. can_options=%s args=%s exported=%s" % (can_options, args, exported))
PY
