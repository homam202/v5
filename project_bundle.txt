import 'dart:math' as math;
import '../core/processing/grid_math.dart';

enum Verdict { realMetal, possibleMetal, realVoid, possibleVoid, looseSoil, reliable, possible, uncertain }

/// One detected target.
class Finding {
  final String kind; // soil | void | moisture | organic | rock | metal
  final double confidence; // 0..1 (an estimate, not a calibrated probability)
  final double r0, c0, r1, c1; // cell bounds in ORIGINAL grid units (r1/c1 exclusive)
  final double peakDelta; // signed extreme deviation from the normal level (counts)
  final double widthCm, lengthCm, depthM;
  final Verdict verdict;
  final Map<String, double> probs; // class distribution for this target
  final String shape; // compact | linear | planar
  final bool edgeCut; // touches the scan border: probably continues outside
  final bool lowRes; // smaller than the scan resolution can describe
  const Finding(this.kind, this.confidence, this.r0, this.c0, this.r1, this.c1, this.peakDelta,
      this.widthCm, this.lengthCm, this.depthM, this.verdict,
      {this.probs = const {}, this.shape = 'compact', this.edgeCut = false, this.lowRes = false});
}

/// Knowledge-based classifier. Typical magnetic signatures:
///  - ferrous metal: strong, compact positive peak, usually with a weak negative side lobe (dipole)
///  - void / cavity: negative anomaly (less magnetic material than the surroundings)
///  - rock: broad, moderate positive anomaly
///  - moisture: broad, weak negative anomaly
/// The weights are hand-set heuristics (not trained on field data); treat results as guidance.
Map<String, double> classifyTarget({
  required double sign,
  required double ratio, // |peak| / threshold
  required int area, // cells
  required double compactness, // area / bounding-box area
  required double oppRatio, // opposite-polarity lobe / peak
}) {
  double sat(double x) => x < 1 ? x : 1;
  final pos = sign > 0;
  final s = <String, double>{
    'metal': pos
        ? 1.2 * sat(ratio / 4) + 0.8 * (area <= 9 ? 1 : (area <= 16 ? 0.5 : 0)) +
            0.6 * sat(oppRatio / 0.3) + 0.4 * (compactness > 0.6 ? 1 : 0)
        : 0.15,
    'rock': pos ? 0.8 * (area > 9 ? 1 : 0.3) + 0.5 * (ratio < 3 ? 1 : 0.3) + 0.4 * (oppRatio < 0.2 ? 1 : 0) : 0.1,
    'void': !pos ? 1.2 * sat(ratio / 3) + 0.6 * (area >= 2 ? 1 : 0.3) + 0.4 * (oppRatio < 0.15 ? 1 : 0) : 0.1,
    'moisture': !pos ? 0.8 * (area > 9 ? 1 : 0.2) + 0.5 * (ratio < 2.5 ? 1 : 0.2) : 0.1,
    'organic': 0.5 * (ratio < 1.8 ? 1 : 0) + 0.4 * (compactness < 0.5 ? 1 : 0) + 0.1,
    'soil': 0.3,
  };
  final e = s.map((k, v) => MapEntry(k, math.exp(2.0 * v)));
  final sum = e.values.reduce((a, b) => a + b);
  return e.map((k, v) => MapEntry(k, v / sum));
}

/// Smart engine: always works, never needs a model.
List<Finding> buildFindings({
  required List<List<double>> m, // values to analyse, no NaN
  required int spacingCm,
  required double thrCounts, // user threshold converted to sensor counts
  required double soilFactor,
  double ntPerCount = 1.0,
  double refDepthCm = 0, // optional depth calibration: known depth ...
  double refAmpNt = 0, // ... at a known amplitude
}) {
  final rows = m.length, cols = m.first.length;
  final flat = [for (final r in m) ...r];
  final normal = median(flat); // "normal soil value"
  final sigma = robustSigma(flat);
  final thr = math.max(thrCounts, 3 * sigma).clamp(1.0, 1e9).toDouble();
  final calibrated = refDepthCm > 0 && refAmpNt > 0;

  double depthAt(int pr, int pc) {
    final peak = m[pr][pc] - normal;
    final sign = peak >= 0 ? 1.0 : -1.0;
    final half = peak.abs() / 2;
    int run(int dr, int dc) {
      var n = 0, r = pr + dr, c = pc + dc;
      while (r >= 0 && r < rows && c >= 0 && c < cols && sign * (m[r][c] - normal) >= half) {
        n++;
        r += dr;
        c += dc;
      }
      return n;
    }

    final w = ((run(0, -1) + run(0, 1) + 1) + (run(-1, 0) + run(1, 0) + 1)) / 2;
    final fwhm = w * spacingCm / 100; // metres
    final byWidth = fwhm / 1.5 * soilFactor; // rough rule of thumb
    if (!calibrated) return byWidth.clamp(0.1, 5.0).toDouble();
    // point-source amplitude falls roughly with 1/z^3
    final ampNt = math.max(peak.abs() * ntPerCount, 1e-6);
    final byAmp = refDepthCm / 100 * math.pow(refAmpNt / ampNt, 1 / 3);
    return (0.5 * byWidth + 0.5 * byAmp).clamp(0.1, 5.0).toDouble();
  }

  Verdict verdictFor(String kind, double conf, double peak) {
    final okMetal = peak > 0.8 * thr, okVoid = peak < -0.8 * thr;
    if (kind == 'metal') {
      if (okMetal) return conf >= 0.6 ? Verdict.realMetal : Verdict.possibleMetal;
      return conf >= 0.35 ? Verdict.looseSoil : Verdict.uncertain;
    }
    if (kind == 'void') {
      if (okVoid) return conf >= 0.6 ? Verdict.realVoid : Verdict.possibleVoid;
      return conf >= 0.35 ? Verdict.looseSoil : Verdict.uncertain;
    }
    return conf >= 0.6 ? Verdict.reliable : (conf >= 0.4 ? Verdict.possible : Verdict.uncertain);
  }

  final out = <Finding>[];
  final seen = List.generate(rows, (_) => List.filled(cols, false));
  for (var r = 0; r < rows; r++) {
    for (var c = 0; c < cols; c++) {
      final dv = m[r][c] - normal;
      if (seen[r][c] || dv.abs() <= thr) continue;
      final sign = dv > 0 ? 1.0 : -1.0;
      final stack = [(r, c)];
      seen[r][c] = true;
      var rMin = r, rMax = r, cMin = c, cMax = c, area = 0, pr = r, pc = c;
      var peak = dv;
      while (stack.isNotEmpty) {
        final (cr, cc) = stack.removeLast();
        area++;
        rMin = math.min(rMin, cr);
        rMax = math.max(rMax, cr);
        cMin = math.min(cMin, cc);
        cMax = math.max(cMax, cc);
        final v = m[cr][cc] - normal;
        if (v.abs() > peak.abs()) {
          peak = v;
          pr = cr;
          pc = cc;
        }
        for (final (dr, dc) in [(1, 0), (-1, 0), (0, 1), (0, -1)]) {
          final nr = cr + dr, nc = cc + dc;
          if (nr < 0 || nr >= rows || nc < 0 || nc >= cols || seen[nr][nc]) continue;
          if ((m[nr][nc] - normal) * sign > thr) {
            seen[nr][nc] = true;
            stack.add((nr, nc));
          }
        }
      }
      final ratio = peak.abs() / thr;
      if (area == 1 && ratio < 2) continue; // lone weak cell = noise

      // opposite-polarity side lobe near the target (dipole signature)
      var opp = 0.0;
      for (var rr = math.max(0, rMin - 2); rr <= math.min(rows - 1, rMax + 2); rr++) {
        for (var cc = math.max(0, cMin - 2); cc <= math.min(cols - 1, cMax + 2); cc++) {
          final o = -(m[rr][cc] - normal) * sign;
          if (o > opp) opp = o;
        }
      }
      final bw = cMax - cMin + 1, bh = rMax - rMin + 1;
      final compactness = area / (bw * bh);
      final probs = classifyTarget(
        sign: sign,
        ratio: ratio,
        area: area,
        compactness: compactness,
        oppRatio: opp / peak.abs(),
      );
      final best = probs.entries.reduce((a, b) => a.value >= b.value ? a : b);
      if (best.key == 'soil') continue; // ambiguous -> treat as background

      final aspect = math.max(bw, bh) / math.min(bw, bh);
      final shape = (aspect >= 3 && math.min(bw, bh) <= 2)
          ? 'linear'
          : ((area >= 16 && compactness > 0.5 && math.min(bw, bh) >= 4) ? 'planar' : 'compact');
      out.add(Finding(best.key, best.value, rMin.toDouble(), cMin.toDouble(), rMax + 1.0, cMax + 1.0, peak,
          bw * spacingCm.toDouble(), bh * spacingCm.toDouble(), depthAt(pr, pc),
          verdictFor(best.key, best.value, peak),
          probs: probs,
          shape: shape,
          edgeCut: rMin == 0 || cMin == 0 || rMax == rows - 1 || cMax == cols - 1,
          lowRes: area <= 1));
    }
  }

  out.sort((a, b) => b.confidence.compareTo(a.confidence));
  return out;
}

import 'dart:math' as math;

int nextPow2(int n) {
  var p = 1;
  while (p < n) {
    p <<= 1;
  }
  return p;
}

/// In-place iterative radix-2 FFT. Length must be a power of two.
void fft1d(List<double> re, List<double> im, {bool inverse = false}) {
  final n = re.length;
  // bit reversal
  var j = 0;
  for (var i = 1; i < n; i++) {
    var bit = n >> 1;
    for (; (j & bit) != 0; bit >>= 1) {
      j ^= bit;
    }
    j ^= bit;
    if (i < j) {
      final tr = re[i];
      re[i] = re[j];
      re[j] = tr;
      final ti = im[i];
      im[i] = im[j];
      im[j] = ti;
    }
  }
  for (var len = 2; len <= n; len <<= 1) {
    final ang = 2 * math.pi / len * (inverse ? 1 : -1);
    final wr = math.cos(ang), wi = math.sin(ang);
    for (var i = 0; i < n; i += len) {
      var cr = 1.0, ci = 0.0;
      for (var k = 0; k < len ~/ 2; k++) {
        final a = i + k, b = i + k + len ~/ 2;
        final xr = re[b] * cr - im[b] * ci;
        final xi = re[b] * ci + im[b] * cr;
        re[b] = re[a] - xr;
        im[b] = im[a] - xi;
        re[a] += xr;
        im[a] += xi;
        final ncr = cr * wr - ci * wi;
        ci = cr * wi + ci * wr;
        cr = ncr;
      }
    }
  }
  if (inverse) {
    for (var i = 0; i < n; i++) {
      re[i] /= n;
      im[i] /= n;
    }
  }
}

int _reflect(int i, int n) {
  final p = 2 * n;
  var k = i % p;
  if (k < 0) k += p;
  return k < n ? k : p - 1 - k;
}

/// Applies a radially symmetric gain g(k) (k in rad/cell) in the frequency domain.
/// The grid is mirror-padded to a power of two to limit edge artefacts.
List<List<double>> spectralGain(List<List<double>> m, double Function(double k) gain) {
  final rows = m.length, cols = m.first.length;
  final pr = nextPow2(rows * 2), pc = nextPow2(cols * 2);
  final or = (pr - rows) ~/ 2, oc = (pc - cols) ~/ 2;
  final re = List.generate(pr, (i) => List.generate(pc, (j) => m[_reflect(i - or, rows)][_reflect(j - oc, cols)]));
  final im = List.generate(pr, (_) => List.filled(pc, 0.0));

  for (var i = 0; i < pr; i++) {
    fft1d(re[i], im[i]);
  }
  for (var j = 0; j < pc; j++) {
    final cr = [for (var i = 0; i < pr; i++) re[i][j]];
    final ci = [for (var i = 0; i < pr; i++) im[i][j]];
    fft1d(cr, ci);
    for (var i = 0; i < pr; i++) {
      re[i][j] = cr[i];
      im[i][j] = ci[i];
    }
  }
  for (var i = 0; i < pr; i++) {
    final ky = 2 * math.pi * (i <= pr ~/ 2 ? i : i - pr) / pr;
    for (var j = 0; j < pc; j++) {
      final kx = 2 * math.pi * (j <= pc ~/ 2 ? j : j - pc) / pc;
      final g = gain(math.sqrt(kx * kx + ky * ky));
      re[i][j] *= g;
      im[i][j] *= g;
    }
  }
  for (var j = 0; j < pc; j++) {
    final cr = [for (var i = 0; i < pr; i++) re[i][j]];
    final ci = [for (var i = 0; i < pr; i++) im[i][j]];
    fft1d(cr, ci, inverse: true);
    for (var i = 0; i < pr; i++) {
      re[i][j] = cr[i];
      im[i][j] = ci[i];
    }
  }
  for (var i = 0; i < pr; i++) {
    fft1d(re[i], im[i], inverse: true);
  }
  return List.generate(rows, (r) => List.generate(cols, (c) => re[r + or][c + oc]));
}

import 'dart:math' as math;

/// Small numeric helpers shared by the filters, the health report and the engine.
double median(List<double> v) {
  if (v.isEmpty) return 0;
  final s = [...v]..sort();
  final n = s.length;
  return n.isOdd ? s[n ~/ 2] : (s[n ~/ 2 - 1] + s[n ~/ 2]) / 2;
}

/// Robust sigma (1.4826 * MAD) of a sample.
double robustSigma(List<double> v) {
  if (v.isEmpty) return 0;
  final m = median(v);
  return 1.4826 * median([for (final x in v) (x - m).abs()]);
}

List<List<double>> copyMatrix(List<List<double>> m) => [for (final r in m) List<double>.from(r)];

/// Median of the 3x3 neighbourhood (edges use the available cells).
List<List<double>> median3x3(List<List<double>> m) {
  final rows = m.length, cols = m.first.length;
  return List.generate(rows, (r) {
    return List.generate(cols, (c) {
      final w = <double>[];
      for (var dr = -1; dr <= 1; dr++) {
        for (var dc = -1; dc <= 1; dc++) {
          final rr = r + dr, cc = c + dc;
          if (rr >= 0 && rr < rows && cc >= 0 && cc < cols) w.add(m[rr][cc]);
        }
      }
      return median(w);
    });
  });
}

/// Robust tilt removal (iterative plane fit that ignores outliers). Keeps the mean level.
/// Returns the corrected matrix; [tilt] (if given) receives a, b in counts per cell.
List<List<double>> removePlane(List<List<double>> m, {List<double>? tilt}) {
  final rows = m.length, cols = m.first.length;
  final xm = (cols - 1) / 2, ym = (rows - 1) / 2;
  var mask = List.generate(rows, (_) => List.filled(cols, true));
  double a = 0, b = 0;

  for (var iter = 0; iter < 4; iter++) {
    double sx = 0, sy = 0, sz = 0, sxx = 0, syy = 0, sxy = 0, sxz = 0, syz = 0, n = 0;
    for (var r = 0; r < rows; r++) {
      for (var c = 0; c < cols; c++) {
        if (!mask[r][c]) continue;
        final x = c - xm, y = r - ym, z = m[r][c];
        n++;
        sx += x;
        sy += y;
        sz += z;
        sxx += x * x;
        syy += y * y;
        sxy += x * y;
        sxz += x * z;
        syz += y * z;
      }
    }
    if (n < 6) break;
    final det = sxx * (syy * n - sy * sy) - sxy * (sxy * n - sy * sx) + sx * (sxy * sy - syy * sx);
    if (det.abs() < 1e-9) break;
    a = (sxz * (syy * n - sy * sy) - sxy * (syz * n - sy * sz) + sx * (syz * sy - syy * sz)) / det;
    b = (sxx * (syz * n - sz * sy) - sxz * (sxy * n - sy * sx) + sx * (sxy * sz - syz * sx)) / det;
    final c0 = (sxx * (syy * sz - sy * syz) - sxy * (sxy * sz - syz * sx) + sxz * (sxy * sy - syy * sx)) / det;

    final res = [
      for (var r = 0; r < rows; r++)
        for (var c = 0; c < cols; c++) m[r][c] - (a * (c - xm) + b * (r - ym) + c0)
    ];
    final med = median(res);
    final sigma = math.max(1.4826 * median([for (final v in res) (v - med).abs()]), 1.0);
    final next = [
      for (var r = 0; r < rows; r++)
        [for (var c = 0; c < cols; c++) (res[r * cols + c] - med).abs() < 2.5 * sigma]
    ];
    if (next.expand((r) => r).where((v) => v).length < 6) break;
    mask = next;
  }
  if (tilt != null) {
    tilt
      ..clear()
      ..addAll([a, b]);
  }
  return [
    for (var r = 0; r < rows; r++) [for (var c = 0; c < cols; c++) m[r][c] - a * (c - xm) - b * (r - ym)]
  ];
}

// Catmull-Rom cubic interpolation on a 4x4 neighbourhood.
double _cr(double t, double p0, double p1, double p2, double p3) =>
    0.5 * ((2 * p1) + (-p0 + p2) * t + (2 * p0 - 5 * p1 + 4 * p2 - p3) * t * t +
        (-p0 + 3 * p1 - 3 * p2 + p3) * t * t * t);

double bicubic(List<List<double>> m, double fy, double fx) {
  final rows = m.length, cols = m.first.length;
  final iy = fy.floor(), ix = fx.floor();
  final ty = fy - iy, tx = fx - ix;
  final col = List<double>.filled(4, 0);
  for (var j = -1; j <= 2; j++) {
    final r = (iy + j).clamp(0, rows - 1);
    double g(int dx) => m[r][(ix + dx).clamp(0, cols - 1)];
    col[j + 1] = _cr(tx, g(-1), g(0), g(1), g(2));
  }
  return _cr(ty, col[0], col[1], col[2], col[3]);
}

/// Bilinear resampling to [rows] x [cols] (used to align a reference scan with the current grid).
List<List<double>> resample(List<List<double>> m, int rows, int cols) {
  final r0 = m.length, c0 = m.first.length;
  if (r0 == rows && c0 == cols) return copyMatrix(m);
  return List.generate(rows, (r) {
    final fy = rows == 1 ? 0.0 : r * (r0 - 1) / (rows - 1);
    return List.generate(cols, (c) {
      final fx = cols == 1 ? 0.0 : c * (c0 - 1) / (cols - 1);
      final iy = fy.floor(), ix = fx.floor();
      final iy1 = math.min(iy + 1, r0 - 1), ix1 = math.min(ix + 1, c0 - 1);
      final ty = fy - iy, tx = fx - ix;
      return (m[iy][ix] * (1 - tx) + m[iy][ix1] * tx) * (1 - ty) + (m[iy1][ix] * (1 - tx) + m[iy1][ix1] * tx) * ty;
    });
  });
}

/// Pearson correlation of two equally sized matrices.
double correlation(List<List<double>> a, List<List<double>> b) {
  final x = [for (final r in a) ...r], y = [for (final r in b) ...r];
  final n = x.length;
  final mx = x.reduce((p, q) => p + q) / n, my = y.reduce((p, q) => p + q) / n;
  var sxy = 0.0, sxx = 0.0, syy = 0.0;
  for (var i = 0; i < n; i++) {
    sxy += (x[i] - mx) * (y[i] - my);
    sxx += (x[i] - mx) * (x[i] - mx);
    syy += (y[i] - my) * (y[i] - my);
  }
  final d = math.sqrt(sxx * syy);
  return d < 1e-9 ? 0 : sxy / d;
}

import 'dart:math' as math;
import 'grid_math.dart';

/// Scan quality report (6 metrics, 0..1 each) with a grade and plain-language diagnostics.
class HealthReport {
  final Map<String, double> metrics; // coverage, saturation, purity, coherence, background, support
  final double score; // 0..1
  final String grade; // E | G | F | P
  final List<String> issues; // dg0..dg6
  final int unscanned, saturated;
  const HealthReport(this.metrics, this.score, this.grade, this.issues, this.unscanned, this.saturated);
}

HealthReport computeHealth({
  required List<List<double>> raw, // NaN = not measured
  required int spacingCm,
  required double thrCounts,
  double satMax = 1023,
}) {
  final rows = raw.length, cols = raw.first.length;
  final total = rows * cols;
  final measured = [for (final r in raw) ...r.where((v) => !v.isNaN)];
  final unscanned = total - measured.length;
  final double coverage = total == 0 ? 0.0 : measured.length / total;
  if (measured.length < 6) {
    return HealthReport(
        {'coverage': coverage, 'saturation': 1, 'purity': 1, 'coherence': 1, 'background': 1, 'support': 1},
        coverage * 0.5, 'P', ['dg1'], unscanned, 0);
  }
  final med0 = median(measured);
  final m = [for (final r in raw) [for (final v in r) v.isNaN ? med0 : v]];

  final saturated = measured.where((v) => v <= 0 || v >= satMax).length;
  final double satFree = (1 - saturated / measured.length).toDouble();

  final med3 = median3x3(m);
  final res = [for (var r = 0; r < rows; r++) [for (var c = 0; c < cols; c++) m[r][c] - med3[r][c]]];
  final sg = math.max(robustSigma([for (final r in res) ...r]), 1.0);
  final spikes = [for (final r in res) ...r].where((v) => v.abs() > 3.5 * sg).length;
  final spikeFrac = spikes / total;
  final double purity = (1 - math.min(1.0, spikeFrac * 5)).toDouble();

  final tilt = <double>[];
  final flat = removePlane(m, tilt: tilt);
  // Striping = row-to-row jumps. The 2nd difference of the row medians ignores smooth real anomalies.
  final diffs = <double>[];
  for (var r = 0; r < rows; r++) {
    for (var c = 0; c + 1 < cols; c++) {
      diffs.add(m[r][c + 1] - m[r][c]);
    }
  }
  final noise = math.max(robustSigma(diffs) / math.sqrt(2), 1.0); // within-row noise
  final offs = [for (final r in flat) median(r)];
  double coherence = 1.0;
  if (rows >= 5) {
    final d2 = [for (var i = 1; i + 1 < rows; i++) (offs[i + 1] - 2 * offs[i] + offs[i - 1]).abs()];
    final stripeSd = median(d2) * 1.4826 / math.sqrt(6);
    final expected = noise / math.sqrt(cols) * 1.25;
    final ratio = stripeSd / math.max(expected, 0.5);
    coherence = (1 - ((ratio - 1.5) / 4).clamp(0.0, 1.0)).toDouble();
  }

  final drift = math.sqrt(math.pow(tilt[0] * cols, 2) + math.pow(tilt[1] * rows, 2));
  final double background = (1 - (drift / math.max(3 * thrCounts, 1)).clamp(0.0, 1.0)).toDouble();

  final normal = median([for (final r in m) ...r]);
  var anomalous = 0, supported = 0;
  for (var r = 0; r < rows; r++) {
    for (var c = 0; c < cols; c++) {
      final dv = m[r][c] - normal;
      if (dv.abs() <= thrCounts) continue;
      anomalous++;
      var ok = false;
      for (final d in const [(1, 0), (-1, 0), (0, 1), (0, -1)]) {
        final rr = r + d.$1, cc = c + d.$2;
        if (rr < 0 || rr >= rows || cc < 0 || cc >= cols) continue;
        if ((m[rr][cc] - normal) * dv > 0 && (m[rr][cc] - normal).abs() > thrCounts) ok = true;
      }
      if (ok) supported++;
    }
  }
  final double support = anomalous == 0 ? 1.0 : supported / anomalous;

  final score = (1.5 * coverage + satFree + purity + coherence + background + support) / 6.5;
  final grade = score >= 0.85 ? 'E' : (score >= 0.7 ? 'G' : (score >= 0.5 ? 'F' : 'P'));

  final issues = <String>[];
  if (unscanned > 0) issues.add('dg1');
  if (saturated > 0) issues.add('dg2');
  if (spikeFrac > 0.03) issues.add('dg3');
  if (coherence < 0.7) issues.add('dg4');
  if (background < 0.7) issues.add('dg5');
  if (spacingCm > 40) issues.add('dg6');
  if (issues.isEmpty) issues.add('dg0');

  return HealthReport({
    'coverage': coverage,
    'saturation': satFree,
    'purity': purity,
    'coherence': coherence,
    'background': background,
    'support': support,
  }, score, grade, issues, unscanned, saturated);
}

/// Processing steps (non-destructive stack, like a filter history).
enum ProcStep { despike, destripe, background, smooth, edgePreserve, analytic, secondDeriv, upward }

List<List<double>> applySteps(List<List<double>> m, List<ProcStep> steps) {
  var cur = m;
  for (final s in steps) {
    cur = applyStep(cur, s);
  }
  return cur;
}

List<List<double>> applyStep(List<List<double>> m, ProcStep s) {
  switch (s) {
    case ProcStep.despike:
      return _despike(m);
    case ProcStep.destripe:
      return _destripe(m);
    case ProcStep.background:
      return _background(m);
    case ProcStep.smooth:
      return gaussianBlur(m, 1.0);
    case ProcStep.edgePreserve:
      return _bilateral(m);
    case ProcStep.analytic:
      return _analytic(m);
    case ProcStep.secondDeriv:
      return spectralGain(m, (k) => k * k);
    case ProcStep.upward:
      return upwardContinue(m, 2.0);
  }
}

/// Upward continuation by [h] cells (smooths and emphasises deeper sources).
List<List<double>> upwardContinue(List<List<double>> m, double h) =>
    h <= 0 ? copyMatrix(m) : spectralGain(m, (k) => math.exp(-h * k));

List<double> _flat(List<List<double>> m) => [for (final r in m) ...r];

List<List<double>> gaussianBlur(List<List<double>> m, double sigma) {
  final rows = m.length, cols = m.first.length;
  final rad = math.max(1, (3 * sigma).ceil());
  final k = [for (var i = -rad; i <= rad; i++) math.exp(-(i * i) / (2 * sigma * sigma))];
  final sum = k.reduce((a, b) => a + b);
  for (var i = 0; i < k.length; i++) {
    k[i] /= sum;
  }
  final tmp = List.generate(rows, (r) => List.generate(cols, (c) {
        var acc = 0.0;
        for (var i = -rad; i <= rad; i++) {
          acc += m[r][(c + i).clamp(0, cols - 1)] * k[i + rad];
        }
        return acc;
      }));
  return List.generate(rows, (r) => List.generate(cols, (c) {
        var acc = 0.0;
        for (var i = -rad; i <= rad; i++) {
          acc += tmp[(r + i).clamp(0, rows - 1)][c] * k[i + rad];
        }
        return acc;
      }));
}

List<List<double>> _despike(List<List<double>> m) {
  final med = median3x3(m);
  final rows = m.length, cols = m.first.length;
  final res = [for (var r = 0; r < rows; r++) [for (var c = 0; c < cols; c++) m[r][c] - med[r][c]]];
  final sg = math.max(robustSigma(_flat(res)), 1.0);
  return [
    for (var r = 0; r < rows; r++)
      [for (var c = 0; c < cols; c++) res[r][c].abs() > 3.5 * sg ? med[r][c] : m[r][c]]
  ];
}

/// Levels each line (row) to the common median, ignoring the overall tilt.
List<List<double>> _destripe(List<List<double>> m) {
  final flat = removePlane(m);
  final gm = median(_flat(flat));
  return [
    for (var r = 0; r < m.length; r++)
      () {
        final off = median(flat[r]) - gm;
        return [for (final v in m[r]) v - off];
      }()
  ];
}

/// Subtracts a smooth regional field (heavy blur of the outlier-clipped data).
List<List<double>> _background(List<List<double>> m) {
  final all = _flat(m);
  final med = median(all);
  final sg = math.max(robustSigma(all), 1.0);
  final clip = [for (final r in m) [for (final v in r) v.clamp(med - 2 * sg, med + 2 * sg).toDouble()]];
  final sigma = math.max(2.0, math.min(m.length, m.first.length) / 4);
  final low = gaussianBlur(clip, sigma);
  return [
    for (var r = 0; r < m.length; r++) [for (var c = 0; c < m.first.length; c++) m[r][c] - low[r][c] + med]
  ];
}

List<List<double>> _bilateral(List<List<double>> m) {
  final rows = m.length, cols = m.first.length;
  final sr = math.max(robustSigma(_flat(m)) * 2.0, 1.0);
  const ss = 1.2;
  return List.generate(rows, (r) => List.generate(cols, (c) {
        var acc = 0.0, wsum = 0.0;
        for (var dr = -2; dr <= 2; dr++) {
          for (var dc = -2; dc <= 2; dc++) {
            final rr = r + dr, cc = c + dc;
            if (rr < 0 || rr >= rows || cc < 0 || cc >= cols) continue;
            final dv = m[rr][cc] - m[r][c];
            final w = math.exp(-(dr * dr + dc * dc) / (2 * ss * ss)) * math.exp(-(dv * dv) / (2 * sr * sr));
            acc += w * m[rr][cc];
            wsum += w;
          }
        }
        return acc / wsum;
      }));
}

/// Analytic-signal amplitude: sqrt(gx^2 + gy^2 + gz^2). Peaks sit over the source edges/centre
/// regardless of the polarity of the anomaly.
List<List<double>> _analytic(List<List<double>> m) {
  final rows = m.length, cols = m.first.length;
  final gz = spectralGain(m, (k) => k);
  return List.generate(rows, (r) => List.generate(cols, (c) {
        final gx = (m[r][math.min(c + 1, cols - 1)] - m[r][math.max(c - 1, 0)]) / 2;
        final gy = (m[math.min(r + 1, rows - 1)][c] - m[math.max(r - 1, 0)][c]) / 2;
        return math.sqrt(gx * gx + gy * gy + gz[r][c] * gz[r][c]);
      }));
}

double _iouF(Finding a, Finding b) {
  double mx(double x, double y) => x > y ? x : y;
  double mn(double x, double y) => x < y ? x : y;
  final ix = mx(0, mn(a.c1, b.c1) - mx(a.c0, b.c0));
  final iy = mx(0, mn(a.r1, b.r1) - mx(a.r0, b.r0));
  final inter = ix * iy;
  final u = (a.c1 - a.c0) * (a.r1 - a.r0) + (b.c1 - b.c0) * (b.r1 - b.r0) - inter;
  return u <= 0 ? 0 : inter / u;
}

  // ------------------------------------------------------------ compare
  Future<void> _importReference() async {
    try {
      final res = await FilePicker.platform.pickFiles(type: FileType.any);
      final path = res?.files.single.path;
      if (path == null) return;
      final m = await ScanStore.readMatrixFile(path);
      final med = median([for (final r in m) ...r.where((v) => !v.isNaN)]);
      s.reference.value = [for (final r in m) [for (final v in r) v.isNaN ? med : v]];
    } catch (e) {
      _snack('Import failed: $e');
    }
  }

  Widget _compareView(BuildContext context, List<List<double>> proc) {
    final ref = s.reference.value;
    if (ref == null) {
      return Center(
        child: Card(
          child: Padding(
            padding: const EdgeInsets.all(16),
            child: Column(mainAxisSize: MainAxisSize.min, children: [
              const Icon(Icons.compare, size: 40, color: TM.cyan),
              const SizedBox(height: 8),
              Text(tr(context, 'Compare with a reference', 'المقارنة مع مرجع'), style: const TextStyle(fontWeight: FontWeight.bold)),
              const SizedBox(height: 4),
              Text(tr(context, 'A reference scan (CSV / JSON) or a simulated survey with known targets.',
                      'مسح مرجعي (CSV / JSON) أو مسح محاكى بأهداف معروفة.'),
                  textAlign: TextAlign.center, style: const TextStyle(color: TM.grey, fontSize: 12)),
              const SizedBox(height: 12),
              FilledButton.icon(onPressed: _importReference, icon: const Icon(Icons.upload_file), label: Text(tr(context, 'Import reference', 'استيراد مرجع'))),
              const SizedBox(height: 8),
              OutlinedButton.icon(onPressed: () => _simulate(), icon: const Icon(Icons.science), label: Text(tr(context, 'Simulate reference', 'محاكاة مرجع'))),
              TextButton(onPressed: () => _simulate(asScan: true), child: Text(tr(context, 'Load a simulated survey as the scan', 'تحميل مسح محاكى كمسح حالي'))),
            ]),
          ),
        ),
      );
    }
    final rows = proc.length, cols = proc.first.length;
    final res = compareMatrices(proc, ref);
    final fs = _findings(proc);
    final fr = buildFindings(
      m: resampleTo(ref, rows, cols),
      spacingCm: s.grid.spacingCm,
      thrCounts: s.thresholdNt.value / s.ntPerCount.value,
      soilFactor: soilTypes[s.soil.value].factor,
    );
    final matched = fr.where((r) => fs.any((x) => x.kind == r.kind && _iouF(x, r) > 0.25)).length;
    final extra = fs.where((x) => !fr.any((r) => r.kind == x.kind && _iouF(x, r) > 0.25)).length;
    final rtl = Directionality.of(context) == TextDirection.rtl;
    final c = res.corr;
    return Column(children: [
      Expanded(
        child: Stack(children: [
          CustomPaint(
              size: Size.infinite,
              painter: HeatmapPainter(rtl, res.diff, const [], palette: s.palette.value, reverse: s.reversePalette.value)),
          PositionedDirectional(
            start: 10, top: 4,
            child: Container(
              padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 2),
              decoration: BoxDecoration(color: TM.navy.withOpacity(0.8), borderRadius: BorderRadius.circular(8)),
              child: Text(tr(context, 'Difference: scan − reference', 'الفرق: المسح − المرجع'), style: const TextStyle(fontSize: 12, color: TM.gold)),
            ),
          ),
        ]),
      ),
      Padding(
        padding: const EdgeInsets.all(8),
        child: Card(
          child: Padding(
            padding: const EdgeInsets.all(12),
            child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
              Row(children: [
                _stat(context, tr(context, 'Correlation', 'الارتباط'), c.toStringAsFixed(2), c > 0.8 ? TM.cyan : (c > 0.5 ? TM.gold : TM.hot)),
                _stat(context, 'RMSE', '${res.rmse.toStringAsFixed(1)}', Colors.white),
                _stat(context, tr(context, 'Targets matched', 'أهداف مطابقة'), '$matched/${fr.length}', matched == fr.length ? TM.cyan : TM.gold),
                _stat(context, tr(context, 'Extra', 'زائدة'), '$extra', extra == 0 ? TM.cyan : TM.gold),
              ]),
              Row(mainAxisAlignment: MainAxisAlignment.end, children: [
                TextButton(onPressed: _importReference, child: Text(tr(context, 'Change', 'تغيير'))),
                TextButton(onPressed: () => _simulate(), child: Text(tr(context, 'Simulate', 'محاكاة'))),
                TextButton(onPressed: () => s.reference.value = null, child: Text(tr(context, 'Clear', 'مسح'))),
              ]),
            ]),
          ),
        ),
      ),
    ]);
  }

    if (fl.isEmpty) return const [];
    final l = fl.first.probs.entries.toList()..sort((a, b) => b.value.compareTo(a.value));

  }

  Future<void> _exportLog(List<Finding> fl) async {
    try {
      final sb = StringBuffer(
          'kind,confidence,verdict,width_cm,length_cm,depth_m,shape,peak_counts,row0,col0,row1,col1,edge_cut,low_res\n');
      for (final f in fl) {
        sb.writeln('${f.kind},${f.confidence.toStringAsFixed(2)},${f.verdict.name},${f.widthCm.round()},'
            '${f.lengthCm.round()},${f.depthM.toStringAsFixed(2)},${f.shape},${f.peakDelta.toStringAsFixed(1)},'
            '${f.r0.toStringAsFixed(0)},${f.c0.toStringAsFixed(0)},${f.r1.toStringAsFixed(0)},${f.c1.toStringAsFixed(0)},'
            '${f.edgeCut},${f.lowRes}');
      }
      final dir = await getTemporaryDirectory();
      final path = p.join(dir.path, 'terramag_targets_${DateTime.now().millisecondsSinceEpoch}.csv');
      await File(path).writeAsString(sb.toString());
      await Share.shareXFiles([XFile(path)], text: 'TerraMag targets');
    } catch (e) {
      _snack('Export failed: $e');
    }
  }

  Future<void> _exportPng() async {
    try {
      final b = _viewKey.currentContext?.findRenderObject() as RenderRepaintBoundary?;
      if (b == null) return;
      final img = await b.toImage(pixelRatio: 2.0);
      final bytes = await img.toByteData(format: ui.ImageByteFormat.png);
      if (bytes == null) return;
      final dir = await getTemporaryDirectory();
      final path = p.join(dir.path, 'terramag_view_${DateTime.now().millisecondsSinceEpoch}.png');
      await File(path).writeAsBytes(bytes.buffer.asUint8List());
      await Share.shareXFiles([XFile(path)], text: 'TerraMag');
    } catch (e) {
      _snack('Export failed: $e');
    }
  }

/// Bilinear resampling to a new size (so a reference of another size can be compared).
List<List<double>> resampleTo(List<List<double>> m, int rows, int cols) {
  final r0 = m.length, c0 = m.first.length;
  if (r0 == rows && c0 == cols) return copyMatrix(m);
  return List.generate(rows, (r) {
    final fy = rows == 1 ? 0.0 : r * (r0 - 1) / (rows - 1);
    final iy = fy.floor().clamp(0, r0 - 1), iy1 = math.min(iy + 1, r0 - 1);
    final ty = fy - iy;
    return List.generate(cols, (c) {
      final fx = cols == 1 ? 0.0 : c * (c0 - 1) / (cols - 1);
      final ix = fx.floor().clamp(0, c0 - 1), ix1 = math.min(ix + 1, c0 - 1);
      final tx = fx - ix;
      return (m[iy][ix] * (1 - tx) + m[iy][ix1] * tx) * (1 - ty) + (m[iy1][ix] * (1 - tx) + m[iy1][ix1] * tx) * ty;
    });
  });
}

double _gauss(math.Random r) {
  final u = 1 - r.nextDouble(), v = r.nextDouble();
  return math.sqrt(-2 * math.log(u)) * math.cos(2 * math.pi * v);
}

/// Synthetic reference survey with a few known targets and Gaussian noise [sigma] (counts).
/// Handy for testing the filters and the detector against a "ground truth".
List<List<double>> simulateReference({
  required int rows,
  required int cols,
  required double sigma,
  int seed = 1,
  double base = 505,
}) {
  final rnd = math.Random(seed);
  final specs = <List<double>>[
    // cx, cy, spread, amplitude
    [0.25 + rnd.nextDouble() * 0.5, 0.2 + rnd.nextDouble() * 0.2, 0.06, 280 + rnd.nextDouble() * 80],
    [0.2 + rnd.nextDouble() * 0.2, 0.6 + rnd.nextDouble() * 0.25, 0.08, -(170 + rnd.nextDouble() * 70)],
    [0.6 + rnd.nextDouble() * 0.25, 0.6 + rnd.nextDouble() * 0.25, 0.12, 110 + rnd.nextDouble() * 50],
  ];
  return List.generate(rows, (r) {
    return List.generate(cols, (c) {
      final x = cols == 1 ? 0.0 : c / (cols - 1), y = rows == 1 ? 0.0 : r / (rows - 1);
      var v = base;
      for (final t in specs) {
        final d2 = (x - t[0]) * (x - t[0]) + (y - t[1]) * (y - t[1]);
        v += t[3] * math.exp(-d2 / (2 * t[2] * t[2]));
      }
      return (v + sigma * _gauss(rnd)).clamp(0.0, 1023.0).toDouble();
    });
  });
}

class CompareResult {

  final double corr; // Pearson correlation
  final double rmse; // counts

CompareResult compareMatrices(List<List<double>> scan, List<List<double>> ref) {
  final rows = scan.length, cols = scan.first.length;
  final r = resampleTo(ref, rows, cols);
  final ms = median([for (final x in scan) ...x]), mr = median([for (final x in r) ...x]);
  final diff = [for (var i = 0; i < rows; i++) [for (var j = 0; j < cols; j++) (scan[i][j] - ms) - (r[i][j] - mr)]];
  final a = [for (final x in scan) ...x], b = [for (final x in r) ...x];
  final ma = a.reduce((p, q) => p + q) / a.length, mb = b.reduce((p, q) => p + q) / b.length;
  double sab = 0, saa = 0, sbb = 0, se = 0;
  for (var i = 0; i < a.length; i++) {
    sab += (a[i] - ma) * (b[i] - mb);
    saa += (a[i] - ma) * (a[i] - ma);
    sbb += (b[i] - mb) * (b[i] - mb);
  }
  for (final row in diff) {
    for (final d in row) {
      se += d * d;
    }
  }
  final corr = (saa <= 0 || sbb <= 0) ? 0.0 : sab / math.sqrt(saa * sbb);
  return CompareResult(diff, corr, math.sqrt(se / a.length));
}

/// The five filter chips of the 3D/4D screen.
class ScanFilters {
  bool errorCorrection = false; // clip outliers (mean +- 2.5 sigma)
  bool signalCorrection = false; // remove tilt (plane detrend)
  bool interpolation = false; // bicubic x2
  bool subdivision = false; // bilinear x2 (finer cells)
  int rotation = 0; // quarter turns
}

/// Returns (processed matrix, upsampling factor).
(List<List<double>>, int) applyFilters(List<List<double>> input, ScanFilters f) {
  var m = input;
  var factor = 1;
  if (f.errorCorrection) m = _clip(m);
  if (f.signalCorrection) m = removePlane(m);
  if (f.interpolation) {
    m = _upsample(m, 2, true);
    factor = 2;
  } else if (f.subdivision) {
    m = _upsample(m, 2, false);
    factor = 2;
  }
  for (var i = 0; i < f.rotation % 4; i++) {
    m = _rot90(m);
  }
  return (m, factor);
}

List<List<double>> _clip(List<List<double>> m) {
  final flat = m.expand((r) => r).toList();
  final mean = flat.reduce((a, b) => a + b) / flat.length;
  final sd = math.sqrt(flat.map((v) => (v - mean) * (v - mean)).reduce((a, b) => a + b) / flat.length);
  final lo = mean - 2.5 * sd, hi = mean + 2.5 * sd;
  return [for (final r in m) [for (final v in r) v.clamp(lo, hi).toDouble()]];
}

List<List<double>> _upsample(List<List<double>> m, int k, bool cubic) {
  final rows = m.length, cols = m.first.length;
  return List.generate(rows * k, (y) {
    final fy = ((y + 0.5) / k - 0.5).clamp(0.0, rows - 1.0).toDouble();
    return List.generate(cols * k, (x) {
      final fx = ((x + 0.5) / k - 0.5).clamp(0.0, cols - 1.0).toDouble();
      if (cubic) return bicubic(m, fy, fx);
      final iy = fy.floor(), ix = fx.floor();
      final iy1 = math.min(iy + 1, rows - 1), ix1 = math.min(ix + 1, cols - 1);
      final ty = fy - iy, tx = fx - ix;
      return (m[iy][ix] * (1 - tx) + m[iy][ix1] * tx) * (1 - ty) +
          (m[iy1][ix] * (1 - tx) + m[iy1][ix1] * tx) * ty;
    });
  });
}

List<List<double>> _rot90(List<List<double>> m) {
  final rows = m.length, cols = m.first.length;
  return List.generate(cols, (c) => List.generate(rows, (r) => m[rows - 1 - r][c]));
}

/// Converts the 1-D stream of readings into a 2-D matrix (NaN = not yet measured).
/// [snake]=true -> Zigzag walking; false -> Parallel (every row same direction).
/// [startRight]=true -> first row starts at the right edge.
class GridProcessor extends ChangeNotifier {
  int rows, cols, spacingCm;
  bool snake, startRight;
  late List<List<double>> matrix;
  int _cursor = 0;
  int version = 0; // bumped on every change (cache key for the analysis)
  final List<(int, int)> _hist = []; // recorded cells, for undo

  GridProcessor({
    required this.rows,
    required this.cols,
    this.spacingCm = 30,
    this.snake = true,
    this.startRight = false,
  }) {
    _alloc();
  }

  @override
  void notifyListeners() {
    version++;
    super.notifyListeners();
  }

  void _alloc() {
    matrix = List.generate(rows, (_) => List.filled(cols, double.nan));
    _cursor = 0;
    _hist.clear();
  }

  void reset() {
    _alloc();
    notifyListeners();
  }

  void configure({int? rows, int? cols, int? spacingCm, bool? snake, bool? startRight}) {
    this.rows = rows ?? this.rows;
    this.cols = cols ?? this.cols;
    this.spacingCm = spacingCm ?? this.spacingCm;
    this.snake = snake ?? this.snake;
    this.startRight = startRight ?? this.startRight;
    _alloc();
    notifyListeners();
  }

  int get captured => _cursor;
  int get total => rows * cols;
  bool get isComplete => _cursor >= total;
  bool get hasData => matrix.any((r) => r.any((v) => !v.isNaN));
  double get fieldWidthM => cols * spacingCm / 100;
  double get fieldHeightM => rows * spacingCm / 100;

  int get cursor => _cursor;
  int get currentLine => math.min(_cursor ~/ cols + 1, rows);

  /// Cell (row, col) that the next reading will fill.
  (int, int) get nextCell {
    final i = math.min(_cursor, total - 1);
    final r = i ~/ cols;
    var c = i % cols;
    final reverse = snake ? (r.isOdd != startRight) : startRight;
    if (reverse) c = cols - 1 - c;
    return (r, c);
  }

  void addReading(double value) {
    if (isComplete) return;
    final (r, c) = nextCell;
    matrix[r][c] = value;
    _hist.add((r, c));
    _cursor++;
    notifyListeners();
  }

  void undoLast() {
    if (_hist.isEmpty) return;
    final (r, c) = _hist.removeLast();
    matrix[r][c] = double.nan;
    _cursor = math.max(0, _cursor - 1);
    notifyListeners();
  }

  void loadMatrix(List<List<double>> m, {int? spacing, int? cursor}) {
    rows = m.length;
    cols = m.first.length;
    matrix = [for (final r in m) List<double>.from(r)];
    if (spacing != null) spacingCm = spacing;
    _cursor = cursor ?? rows * cols;
    _hist.clear();
    notifyListeners();
  }

  /// Synthetic survey for testing the UI and the AI without hardware.
  void fillDemo() {

      return List.generate(cols, (c) {

        return v.clamp(0.0, 1023.0).toDouble();
      });
    });
    loadMatrix(m);
  }

  ScanStats get stats {
    final v = matrix.expand((r) => r).where((x) => !x.isNaN).toList();
    if (v.isEmpty) return const ScanStats(0, 511.5, 1023);
    final lo = v.reduce(math.min), hi = v.reduce(math.max);
    return ScanStats(lo, v.reduce((a, b) => a + b) / v.length, hi);
  }

  /// Unmeasured cells replaced by the average.
  List<List<double>> filled() {
    final avg = stats.avg;
    return [for (final r in matrix) [for (final v in r) v.isNaN ? avg : v]];
  }

/// Data-relative 5-anchor Jet palette. Returns r,g,b in 0..1.
(double, double, double) jetRelative(double v, ScanStats s) {
  final a = [s.min, (s.min + s.avg) / 2, s.avg, (s.avg + s.max) / 2, s.max];
  const cols = [
    (0.0, 0.0, 1.0),
    (0.0, 1.0, 1.0),
    (0.0, 1.0, 0.0),
    (1.0, 1.0, 0.0),
    (1.0, 0.0, 0.0),
  ];
  if (v <= a.first) return cols.first;
  if (v >= a.last) return cols.last;
  for (var i = 1; i < a.length; i++) {
    if (v < a[i]) {
      final span = a[i] - a[i - 1];
      final t = span <= 1e-9 ? 0.0 : (v - a[i - 1]) / span;
      final (r0, g0, b0) = cols[i - 1];
      final (r1, g1, b1) = cols[i];
      return (r0 + (r1 - r0) * t, g0 + (g1 - g0) * t, b0 + (b1 - b0) * t);
    }
  }
  return cols.last;
}

Color jetRelativeColor(double v, ScanStats s) {
  final (r, g, b) = jetRelative(v, s);
  return Color.fromRGBO((r * 255).round(), (g * 255).round(), (b * 255).round(), 1);
}

/// Which display layer a value belongs to (nearest Jet anchor).
Layer layerOf(double v, ScanStats s) {
  final a = [s.min, (s.min + s.avg) / 2, s.avg, (s.avg + s.max) / 2, s.max];
  if (s.max - s.min < 1e-9) return Layer.soil;
  if (v < (a[0] + a[1]) / 2) return Layer.voids;
  if (v < (a[1] + a[2]) / 2) return Layer.moisture;
  if (v < (a[2] + a[3]) / 2) return Layer.soil;
  if (v < (a[3] + a[4]) / 2) return Layer.minerals;
  return Layer.metals;
}

Color _lerpStops(List<List<double>> s, double t) {
  final pos = t.clamp(0.0, 1.0) * (s.length - 1);
  final i = pos.floor().clamp(0, s.length - 2);
  final f = pos - i;
  double mix(int k) => s[i][k] + (s[i + 1][k] - s[i][k]) * f;
  return Color.fromRGBO((mix(0) * 255).round(), (mix(1) * 255).round(), (mix(2) * 255).round(), 1);
}

/// Runs every filter, the health report and the detector on a demo survey and reports problems.
void _selfTest(BuildContext context) {
  final lines = <String>[];
  final g = GridProcessor(rows: 16, cols: 16)..fillDemo();
  final m = g.filled();
  for (final st in ProcStep.values) {
    try {
      final r = applyStep(m, st);
      final ok = r.length == 16 && r.every((row) => row.length == 16 && row.every((v) => v.isFinite));
      lines.add('${ok ? '✓' : '✗'} ${st.name}');
    } catch (e) {
      lines.add('✗ ${st.name}: $e');
    }
  }
  try {
    final h = computeHealth(raw: g.matrix, spacingCm: 30, thrCounts: 50);
    lines.add('✓ health: ${h.grade} ${(h.score * 100).round()}%');
  } catch (e) {
    lines.add('✗ health: $e');
  }
  try {
    final f = buildFindings(m: m, spacingCm: 30, thrCounts: 50, soilFactor: 1);
    lines.add('${f.length >= 2 ? '✓' : '✗'} detector: ${f.length} targets');
  } catch (e) {
    lines.add('✗ detector: $e');
  }
  final allOk = lines.every((l) => l.startsWith('✓'));
  showDialog(
    context: context,
    builder: (ctx) => AlertDialog(
      title: Text(allOk ? tr(ctx, 'Self-test passed', 'نجح الفحص الذاتي') : tr(ctx, 'Self-test found problems', 'الفحص الذاتي وجد مشكلات')),
      content: SingleChildScrollView(child: Text(lines.join('\n'))),
      actions: [TextButton(onPressed: () => Navigator.pop(ctx), child: Text(tr(ctx, 'Close', 'إغلاق')))],
    ),
  );
}

        title: Text(tr(ctx, 'Delete scan?', 'حذف المسح؟')),
        content: Text(e.name),
        actions: [
          TextButton(onPressed: () => Navigator.pop(ctx, false), child: Text(tr(ctx, 'Cancel', 'إلغاء'))),

  Future<void> _save() async {
    if (!s.grid.hasData) {
      _snack(tr(context, 'Nothing to save yet', 'لا توجد بيانات للحفظ'));
      return;
    }
    final name = await _askName(tr(context, 'Save current scan', 'حفظ المسح الحالي'), await store.nextName());
    if (name == null || name.isEmpty) return;
    await store.save(name, s.grid);
    _refresh();
  }

  Future<void> _import() async {
    try {
      final res = await FilePicker.platform.pickFiles(type: FileType.any);
      final path = res?.files.single.path;
      if (path == null) return;
      final raw = await ScanStore.readMatrixFile(path);
      final m = raw; // NaN cells stay "not measured"
      s.grid.loadMatrix(m);
      await store.save(p.basenameWithoutExtension(path), s.grid);
      _refresh();
      _snack(tr(context, 'Scan imported', 'تم استيراد المسح'));
    } catch (e) {
      _snack('Import failed: $e');
    }
  }

enum BleStatus { disconnected, connecting, connected, reconnecting }

/// BLE link to the ESP32 gradiometer.
/// Protocol: ASCII lines, either one number per line (e.g. 512) or "x,y,z" (nT; the field
/// magnitude is used). Optional "BAT:87" lines report the battery.
class BleService {
  // Nordic UART defaults. If your firmware uses other UUIDs, the service falls back
  // to the first characteristic that supports notifications.
  static final Guid txCharUuid = Guid('6E400003-B5A3-F393-E0A9-E50E24DCCA9E');

  final _samples = StreamController<SensorSample>.broadcast();
  final _status = StreamController<BleStatus>.broadcast();
  Stream<SensorSample> get samples => _samples.stream;
  Stream<BleStatus> get status => _status.stream;

  BleStatus current = BleStatus.disconnected;
  final ValueNotifier<int> battery = ValueNotifier<int>(-1);
  bool vectorSeen = false; // true once a 'x,y,z' line was received (values are nT)
  String? connectedName;
  String? lastName;

  BluetoothDevice? _device;
  StreamSubscription? _connSub, _dataSub;
  String _buffer = '';
  bool _wantConnected = false;
  int _retry = 0;

  void _emit(BleStatus s) {
    current = s;
    _status.add(s);
  }

  /// Used by the built-in sensor simulator.
  void injectSample(double v) => _samples.add(SensorSample(v, DateTime.now()));

  // ---- scanning --------------------------------------------------------
  Stream<List<ScanResult>> get results => FlutterBluePlus.scanResults;
  Stream<bool> get scanning => FlutterBluePlus.isScanning;
  Future<void> startScan() =>
      FlutterBluePlus.startScan(timeout: const Duration(seconds: 8));

  // ---- connect / auto-reconnect ---------------------------------------
  Future<void> connect(BluetoothDevice device) async {
    _wantConnected = true;
    _device = device;
    connectedName = device.platformName.isEmpty ? device.remoteId.str : device.platformName;
    lastName = connectedName;
    await _saveLastDevice(device.remoteId.str, connectedName!);
    await _openLink();
  }

  Future<void> autoConnectLast() async {
    final info = await lastDevice();
    if (info == null) return;
    lastName = info['name'];
    await connect(BluetoothDevice.fromId(info['id']!));
  }

  Future<void> _openLink() async {
    final d = _device;
    if (d == null) return;
    try {
      _emit(BleStatus.connecting);
      await FlutterBluePlus.stopScan();
      await d.connect(timeout: const Duration(seconds: 10), autoConnect: false);

      await _connSub?.cancel();
      _connSub = d.connectionState.listen((s) {
        if (s == BluetoothConnectionState.disconnected && _wantConnected) {
          _emit(BleStatus.reconnecting);
          _scheduleReconnect();
        }
      });

      final services = await d.discoverServices();
      BluetoothCharacteristic? ch;
      for (final s in services) {
        for (final c in s.characteristics) {
          if (c.uuid == txCharUuid) ch = c;
        }
      }
      ch ??= services
          .expand((s) => s.characteristics)
          .where((c) => c.properties.notify)
          .firstOrNull;
      if (ch == null) throw Exception('No notify characteristic');
      await ch.setNotifyValue(true);

      await _dataSub?.cancel();
      _dataSub = ch.onValueReceived.listen(_onBytes);
      d.cancelWhenDisconnected(_dataSub!);

      _retry = 0;
      _buffer = '';
      _emit(BleStatus.connected);
    } catch (_) {
      _emit(BleStatus.reconnecting);
      _scheduleReconnect();
    }
  }

  void _scheduleReconnect() {
    if (!_wantConnected) return;
    final delay = Duration(seconds: (1 << _retry.clamp(0, 5)));
    _retry++;
    Timer(delay, () {
      if (_wantConnected) _openLink();
    });
  }

  Future<void> disconnect() async {
    _wantConnected = false;
    await _dataSub?.cancel();
    await _connSub?.cancel();
    await _device?.disconnect();
    _emit(BleStatus.disconnected);
  }

  // ---- parsing ---------------------------------------------------------
  void _onBytes(List<int> bytes) {
    _buffer += utf8.decode(bytes, allowMalformed: true);
    int nl;
    while ((nl = _buffer.indexOf('\n')) >= 0) {
      final line = _buffer.substring(0, nl).trim();
      _buffer = _buffer.substring(nl + 1);
      if (line.isEmpty) continue;
      if (line.startsWith('BAT:')) {
        battery.value = int.tryParse(line.substring(4)) ?? battery.value;
        continue;
      }
      final parts = line.split(RegExp(r'[,;\s]+')).where((p) => p.isNotEmpty).toList();
      if (parts.length == 3) {
        final x = double.tryParse(parts[0]), y = double.tryParse(parts[1]), z = double.tryParse(parts[2]);
        if (x != null && y != null && z != null) {
          vectorSeen = true;
          injectSample(math.sqrt(x * x + y * y + z * z));
        }
      } else if (parts.length == 1) {
        final v = double.tryParse(parts[0]);
        if (v != null && v.isFinite) injectSample(v);
      }
    }
  }

  // ---- last device (JSON file) ----------------------------------------
  Future<File> _cfg() async =>
      File('${(await getApplicationSupportDirectory()).path}/last_device.json');
  Future<void> _saveLastDevice(String id, String name) async =>
      (await _cfg()).writeAsString(jsonEncode({'id': id, 'name': name}));
  Future<Map<String, String>?> lastDevice() async {
    try {
      final f = await _cfg();
      if (!await f.exists()) return null;
      final m = jsonDecode(await f.readAsString()) as Map;
      if (m['id'] == null) return null;
      return {'id': m['id'] as String, 'name': (m['name'] ?? m['id']) as String};
    } catch (_) {
      return null;
    }
  }
}

/// metal = high double beep, void = low long tone.
class TonePlayer {
  final AudioPlayer _p = AudioPlayer();
  late final Uint8List _metal = _wav([[1500, 0.09], [0, 0.05], [1500, 0.09]]);
  late final Uint8List _void = _wav([[320, 0.30]]);

  Future<void> metal() => _play(_metal);
  Future<void> voidTone() => _play(_void);

  Future<void> _play(Uint8List b) async {
    try {
      await _p.stop();
      await _p.play(BytesSource(b));
    } catch (_) {/* audio is best-effort */}
  }

  /// segments: [frequencyHz (0 = silence), seconds]. 16-bit mono PCM WAV, 22.05 kHz.
  static Uint8List _wav(List<List<double>> segs) {
    const rate = 22050;
    final samples = <int>[];
    for (final s in segs) {
      final n = (s[1] * rate).round();
      for (var i = 0; i < n; i++) {
        final fade = math.min(1.0, math.min(i, n - i) / (rate * 0.008)); // click-free edges
        final v = s[0] == 0 ? 0.0 : math.sin(2 * math.pi * s[0] * i / rate) * 0.8 * fade;
        samples.add((v * 32767).round());
      }
    }
    final data = ByteData(44 + samples.length * 2);
    void str(int o, String t) {
      for (var i = 0; i < t.length; i++) {
        data.setUint8(o + i, t.codeUnitAt(i));
      }
    }

    str(0, 'RIFF');
    data.setUint32(4, 36 + samples.length * 2, Endian.little);
    str(8, 'WAVE');
    str(12, 'fmt ');
    data.setUint32(16, 16, Endian.little);
    data.setUint16(20, 1, Endian.little);
    data.setUint16(22, 1, Endian.little);
    data.setUint32(24, rate, Endian.little);
    data.setUint32(28, rate * 2, Endian.little);
    data.setUint16(32, 2, Endian.little);
    data.setUint16(34, 16, Endian.little);
    str(36, 'data');
    data.setUint32(40, samples.length * 2, Endian.little);
    for (var i = 0; i < samples.length; i++) {
      data.setInt16(44 + i * 2, samples[i], Endian.little);
    }
    return data.buffer.asUint8List();
  }

/// Isometric voxel view. Auto-fits the canvas, rotates about the vertical axis,
/// hides layers, and outlines the AI detection in gold.
class IsoGridPainter extends CustomPainter {
  final List<List<double>> m; // already filtered, raw units
  final double depth; // 0.2 .. 2.0
  final bool volumetric;
  final double rotation; // radians
  final double elevation; // camera elevation in radians (~0.61 = isometric, 1.57 = top view)
  final Set<Layer> hidden;
  final Rect? highlight; // in cell units of [m]
  final PaletteId palette;
  final bool reverse;
  final bool wireframe;

  IsoGridPainter(this.m,
      {this.depth = 1,
      this.volumetric = false,
      this.rotation = 0.6,
      this.elevation = 0.61,
      this.hidden = const {},
      this.highlight,
      this.palette = PaletteId.magnetic,
      this.reverse = false,
      this.wireframe = false});

  @override
  void paint(Canvas canvas, Size size) {
    final rows = m.length, cols = m.first.length;
    final flat = m.expand((r) => r).toList();
    final lo = flat.reduce(math.min), hi = flat.reduce(math.max);
    final stats = ScanStats(lo, flat.reduce((a, b) => a + b) / flat.length, hi);
    final span = (hi - lo).abs() < 1e-9 ? 1.0 : hi - lo;
    final colors = ColorScale(palette, reverse, stats);

    final cr = math.cos(rotation), sr = math.sin(rotation);
    final se = math.sin(elevation), ce = math.cos(elevation);
    Offset pu(double x, double y, double z) {
      final rx = x * cr - y * sr, ry = x * sr + y * cr;
      return Offset((rx - ry) * 0.7071, (rx + ry) * 0.7071 * se - z * ce);
    }

    final maxH = 5.0 * depth;
    var minX = 1e9, maxX = -1e9, minY = 1e9, maxY = -1e9;
    for (final x in [-cols / 2, cols / 2]) {
      for (final y in [-rows / 2, rows / 2]) {
        for (final z in [0.0, maxH]) {
          final p = pu(x, y, z);
          minX = math.min(minX, p.dx);
          maxX = math.max(maxX, p.dx);
          minY = math.min(minY, p.dy);
          maxY = math.max(maxY, p.dy);
        }
      }
    }
    final scale = math.min(size.width * 0.94 / (maxX - minX), size.height * 0.92 / (maxY - minY));
    final bc = Offset((minX + maxX) / 2, (minY + maxY) / 2);
    final center = Offset(size.width / 2, size.height / 2);
    Offset P(double x, double y, double z) => center + (pu(x, y, z) - bc) * scale;

    // Floor outline
    final floor = Path()
      ..moveTo(P(-cols / 2, -rows / 2, 0).dx, P(-cols / 2, -rows / 2, 0).dy)
      ..lineTo(P(cols / 2, -rows / 2, 0).dx, P(cols / 2, -rows / 2, 0).dy)
      ..lineTo(P(cols / 2, rows / 2, 0).dx, P(cols / 2, rows / 2, 0).dy)
      ..lineTo(P(-cols / 2, rows / 2, 0).dx, P(-cols / 2, rows / 2, 0).dy)
      ..close();
    canvas.drawPath(floor, Paint()
      ..style = PaintingStyle.stroke
      ..strokeWidth = 1
      ..color = TM.cyan.withOpacity(0.4));

    // Painter's order: far -> near (larger rx+ry is nearer to the viewer).
    final order = <List<num>>[];
    for (var r = 0; r < rows; r++) {
      for (var c = 0; c < cols; c++) {
        final x = c - cols / 2 + 0.5, y = r - rows / 2 + 0.5;
        order.add([r, c, (x * cr - y * sr) + (x * sr + y * cr)]);
      }
    }
    order.sort((a, b) => a[2].compareTo(b[2]));

    void quad(List<Offset> pts, Color color, {double stroke = 0, Color? sc}) {
      final path = Path()..addPolygon(pts, true);
      canvas.drawPath(path, Paint()..color = color);
      canvas.drawPath(path, Paint()
        ..style = PaintingStyle.stroke
        ..strokeWidth = stroke == 0 ? (wireframe ? 0.9 : 0.4) : stroke
        ..color = sc ?? (wireframe ? Colors.white54 : Colors.black26));
    }

    for (final o in order) {
      final r = o[0].toInt(), c = o[1].toInt();
      final v = m[r][c];
      final hid = hidden.contains(layerOf(v, stats));
      final n = ((v - lo) / span).clamp(0.0, 1.0);
      final h = hid ? 0.1 : math.max(0.15, n * maxH);
      final base = hid
          ? Colors.blueGrey.withOpacity(0.18)
          : colors.colorOf(v).withOpacity(wireframe ? 0.12 : (volumetric ? 0.4 : 1.0));
      final x = c - cols / 2, y = r - rows / 2;

      // [x1,y1,x2,y2,nx,ny] for the 4 side faces
      final sides = [
        [x, y, x + 1, y, 0.0, -1.0],
        [x + 1, y, x + 1, y + 1, 1.0, 0.0],
        [x + 1, y + 1, x, y + 1, 0.0, 1.0],
        [x, y + 1, x, y, -1.0, 0.0],
      ];
      for (var i = 0; i < 4; i++) {
        final s = sides[i];
        final nx = s[4], ny = s[5];
        final vis = (nx * cr - ny * sr) + (nx * sr + ny * cr) > 0;
        if (!vis) continue;
        quad([P(s[0], s[1], 0), P(s[2], s[3], 0), P(s[2], s[3], h), P(s[0], s[1], h)],
            Color.lerp(base, Colors.black, i.isEven ? 0.4 : 0.25)!);
      }
      final marked = highlight != null && !hid && highlight!.contains(Offset(c + 0.5, r + 0.5));
      quad([P(x, y, h), P(x + 1, y, h), P(x + 1, y + 1, h), P(x, y + 1, h)], base,
          stroke: marked ? 1.8 : 0, sc: marked ? TM.gold : null);
    }
  }

  @override
  bool shouldRepaint(covariant IsoGridPainter old) => true;
}
